# --- the hidden attribute actually hides ------------------------------------
# Every component styles itself with a class that sets display, and a class
# beats the UA stylesheet's [hidden] rule. An element the code marked hidden
# that still takes up space is the bug that left the update notice sitting
# empty above every script and "Open from the server..." in the File menu of
# a copy with no server. One assertion covers every component at once.
eval (function(){var bad=[];document.querySelectorAll('[hidden]').forEach(function(el){var r=el.getBoundingClientRect();if(r.width>0||r.height>0)bad.push(el.id||el.className);});window.__hiddenButRendered=bad;return bad.join(', ')||'(none)';})()
expect window.__hiddenButRendered.length == 0
expect document.getElementById('update-bar').getBoundingClientRect().height == 0

# the same rule, where it is easiest to regress: the File menu's server items
# are hidden unless a document server answered, and hidden means gone
jsclick #btn-file
sleep 250
eval (function(){var bad=[];document.querySelectorAll('#file-menu [data-file-action]').forEach(function(b){if(b.hidden&&b.getBoundingClientRect().height>0)bad.push(b.getAttribute('data-file-action'));});window.__menuBad=bad;return bad.join(', ')||'(none)';})()
expect window.__menuBad.length == 0
expect document.getElementById('menu-server-open').getBoundingClientRect().height == 0
jsclick #btn-file
sleep 200

# --- Enter / Shift+Enter at the end of a scene -------------------------------
# A <br> inserted at the very end of a block renders no line on its own, so
# insertHTML("<br>") made a break typed at the end of a scene look like
# nothing happened: the field did not grow, the next character landed BEFORE
# the <br>, and once a trailing <br> existed every later Enter was a no-op.
# insertLineBreak() adds the sentinel <br> that gives the new last line a box.
caret .cell-editable[data-field="audioText"]
eval window.__h0 = document.activeElement.getBoundingClientRect().height
press AAA
key shift Enter
expect document.activeElement.getBoundingClientRect().height > window.__h0
press BBB
expect document.activeElement.innerHTML.indexOf('AAA<br>BBB') !== -1
# a second break, on a field that now already ends in one, still breaks
key Enter
key Enter
press CCC
expect document.activeElement.innerHTML.indexOf('AAA<br>BBB<br><br>CCC') !== -1
# and it survives the focusout sanitiser
eval document.activeElement.blur()
sleep 100
expect document.querySelector('.cell-editable[data-field="audioText"]').innerHTML.indexOf('AAA<br>BBB<br><br>CCC') !== -1
