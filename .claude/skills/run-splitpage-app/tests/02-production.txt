# --- pasting a copied tag into a different scene keeps it a live tag -------
# onRootPaste used to force every paste to plain text (even the app's own
# .tag-anchor markup copied from one cell), so a tag never survived a
# copy/paste into another scene. Simulate the clipboard the browser would
# actually hand a paste of a tagged selection: an HTML flavor carrying the
# tag-anchor span, same as sanitizeHtml already accepts everywhere else.
jsclick .add-row-bar [data-action="add-row"]
sleep 500
eval (function(){var cell=document.querySelector('#rows .row:last-child .visual-cell .cell-editable');cell.focus();var dt=new DataTransfer();dt.setData('text/plain','duct tape');dt.setData('text/html','<span class="tag-anchor" data-tag-cat="prop" data-tag-name="duct tape">duct tape</span>');var evt=new ClipboardEvent('paste',{bubbles:true,cancelable:true,clipboardData:dt});cell.dispatchEvent(evt);return true;})()
sleep 500
expect document.querySelector('#rows .row:last-child .visual-cell .tag-anchor[data-tag-name="duct tape"]') != null
sleep 400
expect document.querySelectorAll('#production-list .tag-entity').length == 4

# --- caret lands after a tag that ends the cell's last row (regression) -----
# A tag closed at the very end of a field is an atomic contenteditable="false"
# span with nothing after it — nothing for the caret to land at, natively.
caret #rows .row:nth-child(1) .visual-cell .cell-editable
press  Camera on the [tripod]
sleep 400
expect document.querySelectorAll('#rows .row:nth-child(1) .visual-cell .tag-anchor').length == 1
# Force a re-render (adding a row rebuilds every cell) so this also exercises
# the render-time fix, not just the fresh-insertion caret placement.
jsclick #rows .row:nth-child(2) [data-action="insert-below"]
sleep 300
caret #rows .row:nth-child(1) .visual-cell .cell-editable
press  and dolly
sleep 300

# that the typed text landed after the tag, not lost or inside it, AND that

sleep 300

expect document.querySelector('#rows .row:nth-child(1) .visual-cell .cell-editable').innerHTML.indexOf('​') == -1
