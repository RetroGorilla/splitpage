# Every export actually produces a file, and the breakdown travels in them.
#
# Downloads don't happen in headless, so the blob is intercepted at
# URL.createObjectURL - the single point every export funnels through
# (deliverFile). The <a download> click is stubbed out so nothing is written.

open

eval (function(){window.__blobs=[];var r=URL.createObjectURL.bind(URL);URL.createObjectURL=function(b){window.__blobs.push(b);return r(b);};var c=HTMLAnchorElement.prototype.click;HTMLAnchorElement.prototype.click=function(){if(this.download)return;return c.apply(this,arguments);};return 'hooked';})()

# something to export, including a tag so the breakdown is non-empty
caret #rows .row .audio-cell .cell-editable
press  Bring the [gaffer tape] to the {kitchen}.
sleep 500
expect document.querySelectorAll('#rows .tag-anchor').length == 2

# the docx is a store-only zip: walk it and confirm the XML parses and the
# breakdown table has a Status column
eval (async function(){var buf=new Uint8Array(await window.__blobs[window.__blobs.length-1].arrayBuffer());var dv=new DataView(buf.buffer);var off=0,files={};while(off+4<=buf.length&&dv.getUint32(off,true)===0x04034b50){var cs=dv.getUint32(off+18,true),nl=dv.getUint16(off+26,true),el=dv.getUint16(off+28,true);var nm=new TextDecoder().decode(buf.subarray(off+30,off+30+nl));var d=off+30+nl+el;files[nm]=buf.subarray(d,d+cs);off=d+cs;}var xml=new TextDecoder().decode(files['word/document.xml']);var doc=new DOMParser().parseFromString(xml,'application/xml');window.__docxOk = {parses:!doc.querySelector("parsererror"), hasStatus: xml.indexOf(">Status</w:t>")!==-1}; return true;})()
expect window.__docxOk.parses
expect window.__docxOk.hasStatus
