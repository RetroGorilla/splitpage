# --- teleprompter -----------------------------------------------------------
click #btn-present
sleep 800
expect !document.getElementById('prompter').hidden

# the prompter covers the sticky topbar (#98). The topbar stays laid out
# while presenting, so hit-test the middle of its band: whatever is on top
# there must belong to the prompter. The height check keeps this honest — a
# topbar that is not laid out at all would make the second line pass for the
# wrong reason.
expect document.querySelector('.topbar').getBoundingClientRect().height > 0
expect (function(){var r=document.querySelector('.topbar').getBoundingClientRect();var el=document.elementFromPoint(r.left+r.width/2,r.top+r.height/2);return !!(el&&el.closest('#prompter'));})()
expect document.querySelectorAll('#prompter-content .prompter-line').length >= 2
expect document.querySelectorAll('#prompter-content .w').length > 10

# play toggles to pause and back

sleep 300
expect document.getElementById('pp-play').textContent.indexOf('Pause') !== -1
jsclick #pp-play
sleep 300
expect document.getElementById('pp-play').textContent.indexOf('Play') !== -1

# mirror flips the scroller
jsclick #pp-mirror
sleep 300
expect document.getElementById('prompter-scroll').classList.contains('mirror')
jsclick #pp-mirror
sleep 200

# --- find & replace ---------------------------------------------------------
# Opened from the File menu, which is also the phone's only route to it.
jsclick #btn-file
jsclick #file-menu [data-file-action="find"]
sleep 400
expect !document.querySelector('.findbar').hidden

eval (function(){var f=document.getElementById('find-input');f.value='the';f.dispatchEvent(new Event('input',{bubbles:true}));return true;})()
sleep 600
expect document.querySelector('.find-count').textContent.indexOf('of') !== -1

# replace-all must leave tag spans untouched - the load-bearing invariant
eval window.__tagsBefore = document.querySelectorAll('#rows .tag-anchor').length
eval (function(){var f=document.getElementById('find-input');f.value='today';f.dispatchEvent(new Event('input',{bubbles:true}));var r=document.getElementById('replace-input');r.value='tomorrow';r.dispatchEvent(new Event('input',{bubbles:true}));return true;})()
sleep 500
jsclick #find-replace-all
sleep 800
expect document.querySelectorAll('#rows .tag-anchor').length === window.__tagsBefore
expect document.body.innerText.indexOf('tomorrow') !== -1

# the tag's own words are deliberately NOT replaceable - but the count says
# they are there, rather than a bare "No matches" that reads as a miss
eval (function(){var f=document.getElementById('find-input');f.value='gaffer';f.dispatchEvent(new Event('input',{bubbles:true}));return true;})()

expect document.querySelector('.find-count').classList.contains('none')

# display only: stepping selects nothing inside the tag
eval getSelection().removeAllRanges()
jsclick #find-next

# a plain miss still says just "No matches", with no tooltip left over
eval (function(){var f=document.getElementById('find-input');f.value='qqzzqq';f.dispatchEvent(new Event('input',{bubbles:true}));return true;})()

# --- a match across a formatting boundary (#85) -----------------------------
# Half plain, half bold is two text nodes, and find used to search one text
# node at a time, so it never found the word. It searches the cell's text
# now: found, selected on screen across both, and replaced as one.
jsclick #find-close
sleep 200
caret #rows .row .visual-cell .cell-editable
press zorb
key ctrl KeyB
press lix
key ctrl KeyB
press  now
sleep 500
expect (function(){var c=document.querySelector('#rows .row .visual-cell .cell-editable');var w=document.createTreeWalker(c,4),n,z=null,l=null;while((n=w.nextNode())){if(/zorb$/.test(n.nodeValue))z=n;if(/^lix/.test(n.nodeValue))l=n;}return !!(z&&l&&z!==l&&parseInt(getComputedStyle(l.parentNode).fontWeight,10)>=600&&parseInt(getComputedStyle(z.parentNode).fontWeight,10)<600);})()
key ctrl KeyF
sleep 300
eval (function(){var f=document.getElementById('find-input');f.value='zorblix';f.dispatchEvent(new Event('input',{bubbles:true}));return true;})()
sleep 400
expect document.getElementById('find-count').textContent == 1 of 1
jsclick #find-next
sleep 300
expect getSelection().toString() == zorblix
eval (function(){var r=document.getElementById('replace-input');r.value='quux';r.dispatchEvent(new Event('input',{bubbles:true}));return true;})()
# A person's mouse-down on Replace takes focus off the cell the match was
# selected in; jsclick does not, so blur it first. (With the cell still
# focused, repainting it fires its focusout, which writes the old DOM back
# over the replacement — true before #85 too.)
eval document.activeElement.blur()
jsclick #find-replace
sleep 500
expect document.querySelector('#rows .row .visual-cell .cell-editable').textContent.indexOf('quux') !== -1
expect document.querySelector('#rows .row .visual-cell .cell-editable').textContent.indexOf('zorb') === -1
expect document.getElementById('find-count').textContent == No matches

quit
