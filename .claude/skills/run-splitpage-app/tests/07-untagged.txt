# --- a stale offer must never wrap the wrong words --------------------------
# The offer holds a live Range. Typing into the MIDDLE of the phrase it covers
# makes the range grow to include what was typed, and the rescan that would
# notice is 220ms behind — so for a moment Tab could wrap "gaffXer tape" and
# label it "gaffer tape". Type into the middle and take the offer immediately.
caret #rows .row .visual-cell .cell-editable
press  Also the gaffer tape rig.
sleep 900
expect document.getElementById('tag-suggest').classList.contains('is-visible')
eval (function(){var c=document.querySelector('#rows .row .visual-cell .cell-editable');var w=document.createTreeWalker(c,NodeFilter.SHOW_TEXT,null),n,t=null;while(w.nextNode()){if((w.currentNode.nodeValue||'').indexOf('gaffer tape')!==-1)t=w.currentNode;}var i=t.nodeValue.indexOf('gaffer tape')+3;var r=document.createRange();r.setStart(t,i);r.collapse(true);var s=getSelection();c.focus();s.removeAllRanges();s.addRange(r);return t.nodeValue.slice(i-3,i+8);})()

# --- a shorter name nested inside a longer, rejected one --------------------
# Names go into the alternation longest-first, so "gaffer tape" wins the match
# in "the gaffer tapes the cable" and then fails its trailing word boundary
# against the "s". Rejecting it after the fact could not un-consume it, so the
# "gaffer" sitting inside what was just rejected was never seen at all. The
# boundaries live in the pattern now, so the engine backtracks to the shorter
# alternative at the same position instead.
caret #rows .row .audio-cell .cell-editable
press  Also a [gaffer] on the crew.
sleep 800
expect document.getElementById('production-badge').textContent == 2
