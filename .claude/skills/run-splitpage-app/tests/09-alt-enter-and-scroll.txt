# --- Alt+Enter splits the FOCUSED column only --------------------------------
caret #rows .row .audio-cell .cell-editable
eval (function(){var c=document.querySelector('#rows .row .audio-cell .cell-editable');var t=c.firstChild;var i=t.nodeValue.indexOf('Today');var r=document.createRange();r.setStart(t,i);r.collapse(true);var s=getSelection();c.focus();s.removeAllRanges();s.addRange(r);return i;})()
key alt Enter
sleep 500

# --- a comment's row pointer follows an Alt+Enter split too -----------------
# The writer's plain Enter already has to move a comment anchor with the text
# it's attached to when a split carries it into a new row (see 06-writer.txt);
# splitRowFieldCore() is the shared core behind both, but that sharing is
# exactly the kind of thing a refactor can silently break, so it gets its own
# check here rather than being trusted. The demo's comment is anchored on
# "word for word" - find it wherever it has ended up after the splits above,
# and split its row right before the anchor.
eval (function(){var a=document.querySelector('#rows .cmt-anchor[data-cmt="c1"]');var cell=a.closest('.cell-editable');var r=document.createRange();r.setStartBefore(a);r.collapse(true);var s=getSelection();cell.focus();s.removeAllRanges();s.addRange(r);return a.textContent;})()
key alt Enter
sleep 500
# where the anchored text actually ended up...
eval (function(){var a=document.querySelector('#rows .cmt-anchor[data-cmt="c1"]');var rows=Array.from(document.querySelectorAll('#rows .row'));window.__anchorRow = rows.indexOf(a.closest('.row'));return window.__anchorRow;})()

# --- a large paste that pushes the caret far down the page brings it back ---
# Insert ~60 lines BEFORE the caret in one shot, the way a real paste does
# (onRootPaste routes through the same execCommand("insertText", ...) call).
# A tall cell like this is exactly the case scrollIntoView on the CELL gets
# wrong: centering the element does not put the CARET (sitting well inside
# it) anywhere near the middle of the screen.
eval (function(){var c=document.querySelector('#rows .row:nth-child(1) .audio-cell .cell-editable');c.focus();var r=document.createRange();r.setStart(c.firstChild,0);r.collapse(true);var s=getSelection();s.removeAllRanges();s.addRange(r);return true;})()
eval (function(){document.execCommand('insertText',false, Array(70).join('Filler line to make this cell very tall indeed. '));return true;})()
sleep 1200
expect getSelection().getRangeAt(0).getBoundingClientRect().top >= 0
expect getSelection().getRangeAt(0).getBoundingClientRect().bottom <= innerHeight
