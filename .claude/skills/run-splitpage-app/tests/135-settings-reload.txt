# Settings survive a reload (#181, #180, #194): changed in the dialog, the
# page reloaded, and each one read back — in Settings and where it acts.
# Its own suite because a goto after a busy session can abort the CDP
# socket (SKILL.md).

open /index.html
sleep 400
wait document.querySelectorAll('#rows .row').length > 0
eval (function(){['splitpage-theme','splitpage-clock','splitpage-hide-shot','splitpage-rundown-min','splitpage-script-scale','splitpage-default-wpm','splitpage-prompter-mirror','splitpage-prompter-flip','splitpage-person-color'].forEach(function(k){localStorage.removeItem(k);});localStorage.setItem('splitpage-comment-author','Kim');return true;})()
eval window.__set = function(id,v){var el=document.getElementById(id);el.value=String(v);el.dispatchEvent(new Event('input',{bubbles:true}));el.dispatchEvent(new Event('change',{bubbles:true}));return true;}
eval window.__settingsTest.open()
wait window.__settingsTest.isOpen()
click #settings-modal input[name="settings-theme"][value="dark"]
click #settings-modal input[name="settings-clock"][value="24"]
click #settings-modal label[for="settings-rundown-min"]
click #settings-modal label[for="settings-hide-shot"]
click #settings-modal label[for="settings-prompter-flip"]
eval window.__set('settings-script-scale', 130)
eval window.__set('settings-default-wpm', 170)
click #settings-modal input[name="settings-person-color"][value="#C8DE63"] + .swatch-dot
expect localStorage.getItem('splitpage-person-color') === '#C8DE63'
key Escape

goto /index.html
