# A target length on the phone (#170): the clock chip's readout
# ("01:35 / 02:00 0:25 left") fits the header with nothing wider than the
# screen, the popover opens on screen below the clock with 16px fields and
# finger-sized presets, a tap on a preset sets it, and the compact strip
# (no ruler) keeps the line, its "Target" label and the hatching on its
# 22px blocks. Its own suite: device metrics are set once, before the page
# loads.
#
# Every `eval` is one physical line (the driver splits on newlines).

open
viewport 375 812 mobile
goto /index.html
wait document.querySelectorAll('#rows .row').length > 0
sleep 600
expect matchMedia('(hover: none)').matches
eval (function(){ window.F = window.__splitpageFormats; var doc = {"schema": 2, "title": "Target phone", "wordsPerMinute": 150, "rows": [{"id": "r1", "visualText": "Wide", "audioText": "Every morning the town wakes to the same sound.", "status": "todo", "lengthSeconds": 60}, {"id": "r2", "visualText": "Close", "audioText": "Mara has mended nets on this quay for thirty years.", "status": "ready", "lengthSeconds": 35}], "comments": []}; F.openFile(new File([JSON.stringify(doc)], 'phone.splitpage')); return true; })()
wait F.state().title === 'Target phone' && document.querySelectorAll('#rows .row').length === 2
sleep 400
eval (function(){ window.T = window.__targetTest; window.ov = function(sel){ return document.querySelector('#rundown-overlay ' + sel); }; window.inView = function(el){var r=el.getBoundingClientRect();return r.width>0&&r.height>0&&r.left>=0&&r.top>=0&&r.right<=document.documentElement.clientWidth&&r.bottom<=innerHeight;}; return true; })()
