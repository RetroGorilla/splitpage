# Stage directions for the performer (#167): writing one in the editor.
#
# A direction is the `dir` mark on words of the Audio column
# (src/core/text), stored as <span class="stage-dir"> and shown dim,
# italic and bracketed. Typing "((beat))" makes one when the second ")"
# is typed, and Ctrl+Z gives back the literal "((beat))" — the tag
# shortcuts' rule; single brackets are prose; in the Visual column
# "((…))" stays what was typed. What is typed right after one is
# narration, not more of it. The Direction button (Audio toolbar only)
# makes the selection one and takes it off again. The model, the readers,
# the merge and the formats are src/test/directions.test.mjs; the
# teleprompter, the read time, the Writer and the export option are
# 156-stage-directions-present.txt.
#
# Every `eval` is one physical line (the driver splits on newlines).

open
wait document.querySelectorAll('#rows .row').length > 0
eval (function(){ window.F = window.__splitpageFormats; window.U = window.__undoTest; window.cellA = function(i){ return document.querySelectorAll('#rows .row .audio-cell .cell-editable')[i || 0]; }; window.cellV = function(){ return document.querySelector('#rows .row .visual-cell .cell-editable'); }; window.aud = function(i){ return F.state().rows[i || 0].audioText; }; window.vis = function(){ return F.state().rows[0].visualText; }; window.tb = function(i){ return document.querySelectorAll('#rows .row')[i || 0].querySelector('.audio-cell .mini-toolbar [data-cmd="direction"]'); }; window.selText = function(el, a, b){ var w = document.createTreeWalker(el, NodeFilter.SHOW_TEXT, null); var n, seen = 0, sn = null, so = 0, en = null, eo = 0; while ((n = w.nextNode())) { var len = n.nodeValue.replace(/​/g, '').length; if (!sn && a <= seen + len) { sn = n; so = a - seen; } if (!en && b <= seen + len) { en = n; eo = b - seen; } seen += len; } el.focus(); var r = document.createRange(); r.setStart(sn, so); r.setEnd(en, eo); var s = getSelection(); s.removeAllRanges(); s.addRange(r); return r.toString(); }; window.clearCell = function(el){ el.focus(); var r = document.createRange(); r.selectNodeContents(el); var s = getSelection(); s.removeAllRanges(); s.addRange(r); document.execCommand('delete'); return el.innerHTML; }; window.caretEnd = function(el){ el.focus(); var r = document.createRange(); r.selectNodeContents(el); r.collapse(false); var s = getSelection(); s.removeAllRanges(); s.addRange(r); return true; }; return true; })()
