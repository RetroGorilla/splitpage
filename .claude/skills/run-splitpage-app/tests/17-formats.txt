# The two project formats: .splitpage and lossless Markdown.
#
# Neither file picker can run headless, so this drives the serializers and
# the open path through window.__splitpageFormats (20d-formats.js) — the
# same trick 03-exports.txt plays on URL.createObjectURL, one level up.
# openFile() takes a real File object and runs the whole open path, modal
# and all, so what is tested here is what File > Open actually does.

# __norm strips the two fields a save legitimately changes and sorts every
# key, so "deep-equals" is a string comparison that cannot be fooled by
# property order.
# __norm also drops docId: it is checked explicitly wherever it matters
# (T1/22-doc-id.txt owns the identity rules), and the legacy .html section
# below needs a comparison that tolerates a *different* id on the two sides.
eval (function(){window.F=window.__splitpageFormats;var sortAll=function(v){if(Array.isArray(v))return v.map(sortAll);if(v&&typeof v==='object'){var o={};Object.keys(v).sort().forEach(function(k){o[k]=sortAll(v[k]);});return o;}return v;};window.__norm=function(s){var c=F.migrate(F.clone(s));delete c.rev;delete c.lastEditedAt;delete c.lastEditedBy;delete c.docId;return JSON.stringify(sortAll(c));};return 'hooked';})()
expect typeof window.F.buildSplitpage === 'function'

# handed to someone else, not the document itself, and carrying the id would
# make every such copy read as a duplicate of the original the moment it opens.
