# Connecting to a server: the Servers dialog and its bookmarks, the join-PIN
# dialog, and the launcher's key being taken out of the URL on load.
#
# What is NOT here: anything that needs a real document server with a PIN.
# This harness serves static files itself and cannot start server.js with an
# environment, so the server half lives in `node tools/test-server.mjs`.
#
# The launch fragment has to be present at LOAD time (the app reads it once,
# on the way up), so it goes on `open` itself — a later `goto` to the same
# path with only the fragment changed does not reload the page.

open /index.html#launch=abc123def
sleep 600
expect document.querySelectorAll('#rows .row').length > 0

# --- the launch key is read, kept for this tab, and taken out of the URL ----
# It must not survive in the address bar: a screenshot, a bookmark or a pasted
# link would carry it, and it is the key to the one endpoint that installs.
expect location.hash === ''
expect location.href.indexOf('launch') === -1
expect sessionStorage.getItem('splitpage-launch-token') == abc123def
# ...and never in localStorage, where it would outlive the launch that minted it
expect localStorage.getItem('splitpage-launch-token') === null

# --- the Servers dialog opens from the File menu ---------------------------
# The entry is dispatched by 20e-connect.js's own capture-phase listener, not
# by handleFileMenuAction() - if that wiring breaks, this is what says so.
jsclick #btn-file
expect !document.getElementById('file-menu').hidden
jsclick #file-menu [data-file-action="servers"]
sleep 300
expect !document.getElementById('connect-modal').hidden
expect document.getElementById('file-menu').hidden
expect document.querySelector('#connect-list .connect-empty') !== null

# --- actually laid out, not just an unhidden dialog nested in a hidden overlay
expect document.getElementById('connect-modal').getBoundingClientRect().width > 0
expect document.querySelector('#connect-modal .modal-panel').offsetParent !== null

# --- an address is required ------------------------------------------------
jsclick #connect-modal [data-connect-action="add"]
sleep 200
expect document.getElementById('connect-error').hidden === false
expect document.getElementById('connect-error').textContent.indexOf('address') !== -1

# --- adding a bookmark persists it, scheme and all -------------------------
eval (function(){document.getElementById('connect-name').value='Studio';document.getElementById('connect-url').value='127.0.0.1:9';return 1;})()
jsclick #connect-modal [data-connect-action="add"]
sleep 500
expect JSON.parse(localStorage.getItem('splitpage-server-bookmarks')).length == 1
expect JSON.parse(localStorage.getItem('splitpage-server-bookmarks'))[0].name == Studio
# normalised on the way in: a scheme added, one trailing slash
expect JSON.parse(localStorage.getItem('splitpage-server-bookmarks'))[0].url == http://127.0.0.1:9/
expect document.querySelectorAll('#connect-list .connect-row').length == 1
# the form is cleared, so the next one does not inherit this one's address
expect document.getElementById('connect-url').value === ''

# --- an SSH bookmark stores host and ports, and never a credential ---------
eval (function(){document.getElementById('connect-name').value='Studio Mac';document.getElementById('connect-url').value='http://localhost:8899/';document.getElementById('connect-ssh-toggle').checked=true;document.getElementById('connect-ssh-toggle').dispatchEvent(new Event('change'));document.getElementById('connect-ssh-userhost').value='me@studio.example.com';document.getElementById('connect-ssh-remote').value='8791';document.getElementById('connect-ssh-local').value='8899';return 1;})()
expect document.getElementById('connect-ssh-fields').hidden === false
jsclick #connect-modal [data-connect-action="add"]
sleep 400
expect JSON.parse(localStorage.getItem('splitpage-server-bookmarks')).length == 2
expect JSON.parse(localStorage.getItem('splitpage-server-bookmarks'))[1].ssh.userHost == me@studio.example.com
expect JSON.parse(localStorage.getItem('splitpage-server-bookmarks'))[1].ssh.localPort == 8899
# the stored bookmark has exactly these keys and no room for a secret
expect Object.keys(JSON.parse(localStorage.getItem('splitpage-server-bookmarks'))[1]).sort().join(',') == id,name,ssh,url
expect Object.keys(JSON.parse(localStorage.getItem('splitpage-server-bookmarks'))[1].ssh).sort().join(',') == localPort,remotePort,userHost

# --- the SSH panel writes the command out; it never runs it ---------------
jsclick #connect-list [data-connect-action="ssh-show"]
sleep 300
expect document.getElementById('connect-ssh-out').hidden === false
expect document.getElementById('connect-ssh-cmd').textContent == ssh -N -L 8899:localhost:8791 me@studio.example.com
shot 18-connect-servers

# --- the PIN dialog validates before it asks the network anything ---------
jsclick #connect-list [data-connect-action="pin"]
sleep 300
expect !document.getElementById('pin-modal').hidden
expect document.getElementById('pin-input').value === ''

# --- actually laid out, not just an unhidden dialog nested in a hidden overlay
expect document.getElementById('pin-modal').getBoundingClientRect().width > 0
expect document.querySelector('#pin-modal .modal-panel').offsetParent !== null
expect document.getElementById('pin-input').type == text
expect document.getElementById('pin-input').getAttribute('inputmode') === null
expect document.getElementById('pin-input').placeholder == Join PIN or key
jsclick #pin-modal [data-connect-action="pin-submit"]
sleep 200
expect document.getElementById('pin-error').hidden === false
expect document.getElementById('pin-error').textContent.indexOf('code') !== -1
# a code typed against something that is not a Splitpage server is refused,
# and refused in the dialog rather than by reloading the page
eval (function(){document.getElementById('pin-input').value='1234';return 1;})()
jsclick #pin-modal [data-connect-action="pin-submit"]
wait document.getElementById('pin-error').textContent.indexOf('code first') === -1 && !document.getElementById('pin-error').hidden
expect sessionStorage.getItem('splitpage-server-token') === null
shot 18-connect-pin
jsclick #pin-modal [data-connect-action="pin-cancel"]
sleep 200
expect document.getElementById('pin-modal').hidden

# --- forgetting a server ---------------------------------------------------
jsclick #connect-list [data-connect-action="remove"]
sleep 300
expect JSON.parse(localStorage.getItem('splitpage-server-bookmarks')).length == 1
jsclick #connect-modal [data-connect-action="close"]
sleep 200
expect document.getElementById('connect-modal').hidden

# --- nothing credential-shaped is stored ----------------------------------
# The only keys this feature may ever add are the base, the bookmarks, the
# join token and the launch token. Anything password-, key- or secret-shaped
# appearing here is a bug, not a feature.
expect Object.keys(localStorage).filter(function(k){return /pass|secret|key|cred|ssh|pin/i.test(k);}).length == 0
expect Object.keys(sessionStorage).filter(function(k){return /pass|secret|key|cred|ssh|pin/i.test(k);}).length == 0
expect localStorage.getItem('splitpage-server-token') === null
expect JSON.stringify(JSON.parse(localStorage.getItem('splitpage-server-bookmarks'))).indexOf('1234') === -1
