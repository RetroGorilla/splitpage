# Plain-language tagging (#210, 07b-plain-tags.js): "VFX:" typed in a Visual
# cell starts an open VFX tag, Tab wraps the words typed since, Esc puts the
# keyword back. The convention is Adam's (2026-09-24): Prop, Location, Loc,
# Cast, VFX, Gear, SFX and Music (both Music/SFX), any case, a colon after,
# Visual column only; Tab: the open tag, then the ghost, then the mention
# prompt. The pure parts (every name, the word test, the wrap) are
# src/test/plain-tags.test.mjs; this is the page.
#
# Typing is `press` (one key at a time): the colon has to arrive on its own,
# as it does for a real writer. Every `eval` is one physical line.

open
wait document.querySelectorAll('#rows .row').length > 1
# eight scenes to write in (the demo document has two)
eval (function(){ for (var i = 0; i < 8; i++) document.querySelector('[data-action="add-row"]').click(); document.activeElement.blur(); return document.querySelectorAll('#rows .row').length; })()
wait document.querySelectorAll('#rows .row').length >= 9
eval (function(){ window.U = window.__undoTest; window.F = window.__splitpageFormats; window.cell = function (i, col) { return document.querySelectorAll('#rows .row')[i].querySelector('.' + (col || 'visual') + '-cell .cell-editable'); }; window.st = function (i, col) { return F.state().rows[i][(col || 'visual') + 'Text'].replace(/&nbsp;/g, ' '); }; window.reg = function (cat, name) { return F.state().productionTags.some(function (t) { return t.cat === cat && t.name === name; }); }; window.kw = function () { return document.querySelectorAll('#rows .plain-tag-keyword').length; }; window.hint = function () { var h = document.getElementById('tag-open-hint'); return h.classList.contains('is-visible') ? h.textContent : ''; }; window.hl = function (cat) { var h = CSS.highlights.get('splitpage-open-tag-' + cat); if (!h || !h.size) return null; var r = Array.from(h)[0]; return String(r).split(String.fromCharCode(0x200B)).join(''); }; window.seen = function (el) { return el.innerText.split(String.fromCharCode(0xA0)).join(' ').split(String.fromCharCode(0x200B)).join(''); }; window.tags = function (el) { return Array.from(el.querySelectorAll('.tag-anchor')).map(function (t) { return t.getAttribute('data-tag-cat') + ':' + t.getAttribute('data-tag-name'); }); }; return true; })()
