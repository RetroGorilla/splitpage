# A `.cmt-anchor` span is editable on purpose (a comment must never lock the
# writer out of their own sentence), so the words inside it can drift away
# from the comment's `quotedText` without the anchor itself disappearing.
# This checks the card notices: still jumpable while the span has text, with
# a small red "Changed" tag and the original quote in its tooltip; orphaned
# once the span is edited down to nothing, with a "No longer in script" tag
# after the quote, and the tag alone when the quote was empty (#186).

open

click #btn-comments
sleep 500
expect document.querySelectorAll('.comment-card').length == 1
expect document.querySelector('.comment-card').classList.contains('is-jumpable')
expect document.querySelector('.comment-quote').classList.contains('changed') === false
expect document.querySelector('.comment-quote').classList.contains('orphaned') === false
expect document.querySelector('.comment-quote').textContent.indexOf('word for word') !== -1

# --- edit the anchored words: drift the live span text away from quotedText -
caret #rows .row:nth-child(2) .audio-cell .cell-editable
eval (function(){var mark=document.querySelector('.cmt-anchor mark');var t=mark.firstChild;var r=document.createRange();r.setStart(t,t.nodeValue.length);r.setEnd(t,t.nodeValue.length);var s=getSelection();s.removeAllRanges();s.addRange(r);return t.nodeValue;})()
type -ish
sleep 300
eval document.activeElement.blur()
sleep 300
expect document.querySelector('.cmt-anchor').textContent.indexOf('word for word-ish') !== -1
