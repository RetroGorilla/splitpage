# The desktop bridge (20f-desktop.js), driven without a real Electron shell.
#
# There is no pre-load hook in this harness that can inject a real
# window.splitpageDesktop before the app script runs — DESKTOP (00-prelude.js)
# is captured once, at load, before any fragment's own top-level code runs.
# So this drives the real code paths a different way: window.__desktopTest
# (20f-desktop.js) swaps the module-scope DESKTOP variable for an in-memory
# fake backend, and everything below — saveDoc, the on-disk-conflict guard,
# the native-menu-command dispatcher — runs exactly the code that would run
# for a real Electron main process, just talking to a fake in place of the
# IPC bridge. Every `eval` here is one physical line — the driver splits a
# script file on newlines, so a multi-line JS block would be read back as one
# command per line and fail with "unknown command". DESKTOP is null again
# (see the last line) once this suite is done, same as every other suite's boot.

expect window.__rtOk

# ---- main's duplicate-id verdict is applied here (DOC-CLAIMS) ------------
# Two copies of one file open side by side share one id; main picks the loser
# and sends splitpage:reassign-doc-id. The page adopts the new id and logs the
# history copy — and, per orchestrator override §1, writes nothing.
eval (function(){ window.__dupFrom = F.docId(); window.__dupTo = 'd' + 'e'.repeat(32); window.fake3._fireReassignDocId({fromId: window.__dupFrom, toId: window.__dupTo, otherTitle: 'X'}); return F.docId(); })()
expect F.docId() === window.__dupTo
expect F.docIdEvents().filter(function(e){ return e.type === 'copy' && e.from === window.__dupFrom && e.to === window.__dupTo; }).length === 1
expect F.writtenOut() === false

# ---- version history goes to the DESKTOP store, not IndexedDB ------------
# makeFakeBackend({history:true}) adds the seven HISTORY-IPC methods
# (20f-desktop.js), so resolveHistoryBackend (03a-history-store.js) picks the
# desktop backend the moment DESKTOP.historyPut exists. The capture rides a
# real caller — loadFileIntoEditor's "Before opening X" push, reached by
# opening a plain .txt — because pushSnapshot lives inside the build's one
# closure and is not exposed. loadFileIntoEditor KEEPS state.docId, so the
# snapshot lands under the document that is still on screen.
