# Sound cues in the Audio column (#219).
#
# "SFX:" or "Music:" starting a line of an Audio cell makes that whole line
# a cue: the `cue` mark (04c-doc-model.js), stored as
# <span class="audio-cue">, shown small, after a note, in the Music/SFX
# colour. It turns into one when its colon is typed, and one Ctrl+Z gives
# the plain words back. Typing on is more of the cue; Enter at its end
# starts an ordinary line. Mid-sentence, and in the Visual column, the
# words are words. The teleprompter never shows one, the read time leaves
# it out, and it is an item of the Production panel's Music/SFX list. The
# model, the readers, the merge and the formats are src/test/cues.test.mjs.
#
# Every `eval` is one physical line (the driver splits on newlines).

open
wait document.querySelectorAll('#rows .row').length > 0
eval (function(){ window.F = window.__splitpageFormats; window.cellA = function(i){ return document.querySelectorAll('#rows .row .audio-cell .cell-editable')[i || 0]; }; window.cellV = function(){ return document.querySelector('#rows .row .visual-cell .cell-editable'); }; window.aud = function(i){ return F.state().rows[i || 0].audioText; }; window.vis = function(){ return F.state().rows[0].visualText; }; window.clearCell = function(el){ el.focus(); var r = document.createRange(); r.selectNodeContents(el); var s = getSelection(); s.removeAllRanges(); s.addRange(r); document.execCommand('delete'); return el.innerHTML; }; window.caretEnd = function(el){ el.focus(); var r = document.createRange(); r.selectNodeContents(el); r.collapse(false); var s = getSelection(); s.removeAllRanges(); s.addRange(r); return true; }; window.rd = function(i){ return document.querySelectorAll('#rows .row .row-read')[i]; }; return true; })()
