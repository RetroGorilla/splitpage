# A side panel on a window under 1100px, and the tagging reference on a
# short one (bug hunt, 2026-09-25).
#
# Below 1100px a panel lies over the script instead of pushing it over.
#   * The page's scroll bar still goes while a panel is open (#187), but
#     its width is handed back (--page-gutter, 03b-side-panels.js; app.css):
#     the toolbar does not step sideways under the pointer and the script
#     does not rewrap (at 900px it used to move the read row 58px).
#   * The rundown strip ends at the panel's left edge, as it does from
#     1100px (#189): the script's length (#207) and the minimize chevron
#     (#178) used to be under the panel's head, now they are the element
#     at their own centres, and a real click on the length opens its
#     popover. Minimized (the thin bar) the same.
#   * The tagging reference card (#205) on a 700px-tall window stays clear
#     of the whole header, the rundown strip included (it reached up over
#     the strip, the length label under its close button), condensed and
#     minimized too, and its "VFX: glow" row keeps its category on the
#     row's own line.
#
# Every `eval` is one physical line.

open
viewport 1000 700
goto /index.html
wait document.querySelectorAll('#rows .row').length > 0
eval (function(){ var d = document.getElementById('desktop-app-bar-dismiss'); if (d) d.click(); var words = function(k){ var a = []; for (var i = 0; i < k; i++) a.push('word' + (i % 7)); return a.join(' '); }; var rows = []; for (var i = 0; i < 60; i++) rows.push({ id: 'r' + i, visualText: 'B-roll: shot ' + i, audioText: 'Scene ' + i + ' ' + words(8 + (i * 7) % 40), status: i % 3 === 0 ? 'shot' : i % 3 === 1 ? 'ready' : 'todo' }); window.__splitpageFormats.openFile(new File([JSON.stringify({ schema: 2, title: 'The Extraordinarily Long Script Title 40', rows: rows })], 'narrow.splitpage')); return true; })()
wait document.querySelectorAll('#rundown .rd-block').length === 60
sleep 400
eval (function(){ var W = window; W.__snap = function(){ var p = document.getElementById('btn-present').getBoundingClientRect(); var r = document.querySelector('#rows .row[data-id="r16"]'); var a = r.querySelector('.cell-editable[data-field="audioText"]').getBoundingClientRect(); return { present: Math.round(p.left), rowTop: Math.round(r.getBoundingClientRect().top), audioW: Math.round(a.width), audioH: Math.round(a.height), sy: Math.round(scrollY) }; }; W.__hits = function(el){ var r = el.getBoundingClientRect(); var h = document.elementFromPoint(r.left + r.width / 2, r.top + r.height / 2); return !!h && el.contains(h); }; return true; })()
expect true
