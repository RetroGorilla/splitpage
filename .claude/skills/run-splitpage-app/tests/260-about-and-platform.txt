# About Splitpage is a dialog in the app's own look, not a native alert()
# (20a-update.js showAbout), and the "get the desktop app" notice is offered
# to a Windows browser only (desktopAppOffered). The Help contents list shows
# a heading's formatting, not its raw markdown (build.mjs mdToHtml), and the
# built page has none of the shell's HTML comments (build.mjs stripHtmlComments).

open

# --- About: a modal, no alert, no command-line advice -------------------------
eval (function(){ window.__alerts = 0; window.alert = function(){ window.__alerts++; }; return true; })()
expect document.getElementById('about-modal').hidden == true
jsclick #btn-file
jsclick #file-menu [data-file-action="about"]
sleep 200
expect document.getElementById('about-modal').hidden == false
expect window.__alerts == 0
expect document.getElementById('about-version').textContent.indexOf('Version ') === 0
expect document.getElementById('about-modal').textContent.indexOf('node update.mjs') === -1
expect document.getElementById('about-modal').textContent.indexOf('start.bat') === -1
expect document.activeElement.hasAttribute('data-about-close')
key Escape
sleep 200
expect document.getElementById('about-modal').hidden == true

# --- the desktop-app notice is for Windows only --------------------------------
# shown exactly when this browser is a Windows desktop one: the harness runs
# on Linux in the cloud and on Windows on Adam's PC, where headless Edge IS
# a Windows browser (2026-09-28). Suite 30 makes each side happen on purpose,
# through the start screen's footer line.
expect document.getElementById('desktop-app-bar').hidden === !(/Windows/.test(navigator.userAgent) && !/Mobi|Android/.test(navigator.userAgent))

# --- Help contents: a `code` heading renders as code, no visible backticks ------
jsclick #btn-file
jsclick #file-menu [data-file-action="help"]
sleep 300
expect document.getElementById('help-toc').textContent.indexOf('`') === -1

# --- the shipped page carries none of src/page.html's editing notes -----------
# (build.mjs stripHtmlComments; the page is served, and saved, verbatim)
expect (function(){ var w = document.createTreeWalker(document, NodeFilter.SHOW_COMMENT); return w.nextNode() === null; })()

# --- the browser tab is named for the script ----------------------------------
# (05-render.js syncPageTitle; a script still on its default name stays plain)
key Escape
sleep 200
expect document.title == Splitpage
eval (function(){ var t = document.getElementById('doc-title'); t.value = 'Keepwarm Mug'; t.dispatchEvent(new Event('input', { bubbles: true })); return true; })()
expect document.title == Keepwarm Mug — Splitpage
eval (function(){ var t = document.getElementById('doc-title'); t.value = 'Untitled Script'; t.dispatchEvent(new Event('input', { bubbles: true })); return true; })()
expect document.title == Splitpage

# --- fields that had only a placeholder are labelled; the notices close with an icon
expect ['import-textarea', 'gdoc-url-input', 'drive-folder-input', 'tag-name'].every(function(id){ return !!document.getElementById(id).getAttribute('aria-label'); })
expect [].every.call(document.querySelectorAll('[data-role="link-input"], [data-role="video-input"]'), function(el){ return !!el.getAttribute('aria-label'); })
expect document.querySelectorAll('.update-bar-dismiss svg').length == 2
