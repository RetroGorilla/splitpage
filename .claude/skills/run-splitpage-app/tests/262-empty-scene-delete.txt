# An empty scene is deleted by one click on its Delete (Adam, 2026-09-28);
# anything else still arms first and deletes on the second click
# (06-rows.js armDelete, "empty" as 05g-scene-clip.js isBlankScene says:
# no words in either column, no pictures or videos, no set length, no
# comments on it). Text typed but not yet synced counts as words. The
# one-click delete is one undo step like any other.
# Every `eval` is one physical line.

eval (function(){ window.F = window.__splitpageFormats; window.U = window.__undoTest; var doc = { schema: 2, title: 'Empties', rows: [ { id: 's1', visualText: 'Wide', audioText: 'Hello.', status: 'todo' }, { id: 'e1', visualText: '', audioText: '', status: 'todo' }, { id: 'e2', visualText: '   ', audioText: '<br>', status: 'ready' }, { id: 'l1', visualText: '', audioText: '', status: 'todo', lengthSeconds: 12 }, { id: 'c1', visualText: '', audioText: '', status: 'todo' }, { id: 'u1', visualText: '', audioText: '', status: 'todo' }, { id: 's2', visualText: 'Close', audioText: 'Bye.', status: 'todo' } ], comments: [ { id: 'k1', rowId: 'c1', field: 'audioText', quotedText: '', author: 'Ann', body: 'Something goes here', createdAt: '2026-09-01T10:00:00Z', kind: 'scene', replies: [] } ] }; window.__loaded = F.state(); F.openFile(new File([JSON.stringify(doc)], 'empties.splitpage')); return true; })()

eval (function(){ window.del = function(id){ return document.querySelector('#rows .row[data-id="' + id + '"] [data-action="delete-row"]'); }; window.has = function(id){ return F.state().rows.some(function(r){ return r.id === id; }); }; window.ids = function(){ return F.state().rows.map(function(r){ return r.id; }).join(','); }; return ids(); })()
