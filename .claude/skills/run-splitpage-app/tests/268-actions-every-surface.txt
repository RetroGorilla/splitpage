# One action, every surface (4.0 milestone 1): Hide shot scenes, Minimise the
# rundown and Version history do the same thing from the File menu, the
# command menu, their shortcut and their own button, and are refused the same
# way where they are not allowed. Written before the three moved onto the
# action registry, and kept unchanged after.

open
viewport 1400 900
eval (function(){ localStorage.removeItem('splitpage-hide-shot'); localStorage.removeItem('splitpage-rundown-min'); window.C = window.__commandTest; window.__cmd = function (q, id) { var hit = null; C.results(q).forEach(function (g) { g.items.forEach(function (it) { if (it.id === id) hit = it; }); }); return hit; }; window.__toasts = function () { return Array.prototype.map.call(document.querySelectorAll('#toast-host .toast'), function (t) { return t.textContent; }).join('|'); }; window.__clearToasts = function () { document.getElementById('toast-host').innerHTML = ''; return true; }; return true; })()

expect __cmd('hide shot', 'toggle-hide-shot').disabled == false
expect __cmd('minimise rundown', 'toggle-rundown-min').label == Minimise the rundown
expect __cmd('version history', 'history').label == Version history…
expect __cmd('version history', 'history').disabled == false
