# Export, Search commands and Find from every surface (4.0 milestone 1): the
# File menu, the command menu, the shortcut and the header button do the same
# thing, and each is refused under the Writer and under a dialog. Written
# before the three moved onto the action registry; the section on the open
# command menu was added with the move.

open
viewport 1400 900
eval (function(){ window.__shown = function (id) { var e = document.getElementById(id); return !!e && !e.hidden; }; window.__settled = function () { return document.getAnimations().every(function (a) { return a.playState === 'finished' || a.playState === 'idle' || a.effect.getTiming().iterations === Infinity; }); }; return true; })()

# --- Export: shortcut, File menu, command menu -----------------------------
key ctrl KeyE
wait __shown('export-modal')
key Escape
wait !__shown('export-modal')

click #btn-file
wait __shown('file-menu')
jsclick #file-menu [data-file-action="export"]
wait __shown('export-modal')
expect document.getElementById('file-menu').hidden
key Escape
wait !__shown('export-modal')

key ctrl KeyK
wait document.getElementById('cmdk').contains(document.activeElement)
type export…
wait !!document.querySelector('#cmdk-list .cmdk-item[data-cmd="export"]')
eval (function(){ var it = document.querySelector('#cmdk-list .cmdk-item[data-cmd="export"]'); if (it) it.click(); return !!it; })()
wait __shown('export-modal')
expect !__shown('cmdk')
key Escape
wait !__shown('export-modal')
