# Save, Save a copy and Open from every surface (4.0 milestone 1): the
# header and Writer Save buttons, the File menu, the command menu and the
# shortcuts reach the same file picker, and the shortcuts keep the browser's
# own Save and Open dialogs away even when the app refuses. The pickers are
# replaced so nothing waits on a real one. Written before the three moved onto
# the action registry; the check that the header's Save explains a refusal
# under a dialog was added with the move.

open
viewport 1400 900
eval (function(){ window.__picked = []; window.showSaveFilePicker = function () { window.__picked.push('save'); return Promise.reject(new DOMException('cancelled', 'AbortError')); }; window.showOpenFilePicker = function () { window.__picked.push('open'); return Promise.reject(new DOMException('cancelled', 'AbortError')); }; window.__prevented = null; window.addEventListener('keydown', function (e) { window.__prevented = e.defaultPrevented; }); window.__take = function () { var p = window.__picked.join(','); window.__picked = []; return p; }; window.__cmdRun = function (q, id) { var it = document.querySelector('#cmdk-list .cmdk-item[data-cmd="' + id + '"]'); if (it) it.click(); return !!it; }; return true; })()
