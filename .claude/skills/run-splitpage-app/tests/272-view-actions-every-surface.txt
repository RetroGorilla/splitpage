# The header's view buttons and the command menu (4.0 milestone 1): the four
# side panels open and close, Write and Present open, the same way from
# either surface. Written before they moved onto the action registry, and
# kept unchanged after.

open
viewport 1400 900
eval (function(){ window.__shown = function (id) { var e = document.getElementById(id); return !!e && !e.hidden; }; window.__run = function (id) { var it = document.querySelector('#cmdk-list .cmdk-item[data-cmd="' + id + '"]'); if (it) it.click(); return !!it; }; return true; })()

# --- each panel: its button opens and closes it; the command menu toggles it -
jsclick #btn-production
sleep 300
expect __shown('production-panel')
expect document.getElementById('btn-production').getAttribute('aria-pressed') == true
