# The Help group from the File menu and the command menu (4.0 milestone 1):
# Settings, Servers, Help, Tour, Set up, About and the tagging reference open
# the same thing from either surface and close the File menu. Written before
# they moved onto the action registry, and kept unchanged after.

open
viewport 1400 900
eval (function(){ window.__shown = function (id) { var e = document.getElementById(id); return !!e && !e.hidden; }; window.__run = function (id) { var it = document.querySelector('#cmdk-list .cmdk-item[data-cmd="' + id + '"]'); if (it) it.click(); return !!it; }; return true; })()

# --- Settings ----------------------------------------------------------------
click #btn-file
sleep 200
jsclick #file-menu [data-file-action="settings"]
wait __shown('settings-modal')
expect document.getElementById('file-menu').hidden
key Escape
wait !__shown('settings-modal')
key ctrl KeyK
sleep 200
type settings
sleep 200
eval __run('settings')
wait __shown('settings-modal')
key Escape
wait !__shown('settings-modal')
