# Keyboard details the action registry has to keep (4.0 milestone 1): Find
# answers Ctrl+F with Shift or Alt held too, Hide shot scenes Ctrl+Alt+H with
# Shift held too; Ctrl+S and Ctrl+Shift+S close an open File menu as they
# save; and with the command menu open, Ctrl+F and Ctrl+Alt+H still act while
# Ctrl+E, Ctrl+S and Ctrl+O do not. Passes against the code before the
# registry (4.0-development 19c66f7) and after it.

eval (function(){ localStorage.removeItem('splitpage-hide-shot'); window.__shown = function (id) { var e = document.getElementById(id); return !!e && !e.hidden; }; window.__picked = []; window.showSaveFilePicker = function () { window.__picked.push('save'); return Promise.reject(new DOMException('cancelled', 'AbortError')); }; window.showOpenFilePicker = function () { window.__picked.push('open'); return Promise.reject(new DOMException('cancelled', 'AbortError')); }; window.__take = function () { var p = window.__picked.join(','); window.__picked = []; return p; }; window.__prevented = null; window.addEventListener('keydown', function (e) { window.__prevented = e.defaultPrevented; }); return true; })()
