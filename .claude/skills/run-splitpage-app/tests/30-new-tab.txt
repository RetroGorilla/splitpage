# The New Tab page (src/js/20j-newtab.js): what a blank tab lands on instead
# of the seeded demo. The page showing itself over a pristine copy and not
# over a document, the two choices ("New script" giving a genuinely empty
# document, "Open a file" loading the picked file into this tab), Esc
# leaving the demo alone underneath, and the harness suppression every other
# suite depends on.
#
# ?newtab=1 is the opt-in, exactly like ?wizard=1 for the first-run wizard:
# both drivers turn this page off for every other suite before any page
# script runs.

# --- the harness suppression every other suite depends on -----------------

# A canonical form of a document, keys sorted and its identity left out, to
# compare what is on screen after the app has had its idle passes over it
# (the b-roll list's read, the registry heal) with a fresh copy of the
# starter: equal means opening one changed nothing about it.
eval (function(){ window.__canon = function (o) { return JSON.stringify(o, function (k, v) { if (v && typeof v === 'object' && !Array.isArray(v)) { var out = {}; Object.keys(v).sort().forEach(function (key) { if (key !== 'docId') out[key] = v[key]; }); return out; } return v; }); }; window.__ids = [window.__splitpageFormats.state().docId]; return true; })()
