# Desktop menu commands that open a dialog, during the tour and over another
# dialog (#307). The tour sits above dialogs, so a dialog opened during it
# opened underneath it; and a second dialog opened over the first. Each such
# command is now refused there with a toast that says what to close first
# (the action's availability, OPENS_A_DIALOG in src/legacy/screen.ts). Help
# still opens during the tour, which it suspends, but not over a dialog. The
# setup wizard still closes itself when a File menu item is chosen, so a
# command chosen over it runs. __dialogs() lists the open dialogs, and
# __tidy() clears the toasts.

open
viewport 1400 900
eval (function(){ window.D = window.__desktopTest; window.__shown = function (id) { var e = document.getElementById(id); return !!e && !e.hidden; }; window.__dialogs = function () { return Array.prototype.map.call(document.querySelectorAll('.modal-backdrop:not([hidden])'), function (m) { return m.id; }).sort().join(','); }; window.__toasts = function () { var t = Array.prototype.map.call(document.querySelectorAll('#toast-host .toast'), function (x) { return x.textContent; }).join('|'); document.getElementById('toast-host').innerHTML = ''; return t; }; window.__run = function (cmd) { D.runMenuCommand(cmd); return true; }; window.__tidy = function () { __toasts(); return true; }; return !!(D && D.runMenuCommand); })()
