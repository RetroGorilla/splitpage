# Present's letter shortcuts and modifier keys (#309). M mirrors the text and F
# flips it, but the prompter's keydown listener looked only at the letter, so
# Ctrl+F (Find, and the browser's own Find) flipped the text as well, and so
# did Ctrl+M and Alt+F. The letters now do nothing with Ctrl, Meta or Alt held.
# Shift stays allowed, as the listener has always taken an upper-case F or M.
# Space is guarded the same way (the arrow keys are left as they were), and
# none of Present's keys act on a key typed into an input, so Ctrl+F then
# typing in the Find box leaves the text alone and keeps the space. The driver
# never sends Meta or an upper-case letter with a modifier, so the matrix
# dispatches KeyboardEvents itself. Every `eval` is one physical line.

open

click #btn-present
sleep 800
expect !document.getElementById('prompter').hidden
expect !document.getElementById('prompter-scroll').classList.contains('flip')
expect !document.getElementById('prompter-scroll').classList.contains('mirror')

# --- every key against all 16 Ctrl/Meta/Alt/Shift combinations --------------------
# M, F and Space act on the prompter only when Ctrl, Meta and Alt are all up
# (Shift either way), and Space calls preventDefault only when it acts.
eval (function(){ var bad = []; var sc = document.getElementById('prompter-scroll'); var pl = document.getElementById('pp-play'); var btns = ['pp-mirror', 'pp-flip', 'pp-play']; var st = function () { return [sc.classList.contains('mirror'), sc.classList.contains('flip'), pl.textContent.indexOf('Pause') !== -1]; }; var keys = [['m', 0], ['M', 0], ['f', 1], ['F', 1], [' ', 2]]; keys.forEach(function (k) { for (var c = 0; c < 16; c++) { var init = { key: k[0], ctrlKey: !!(c & 1), metaKey: !!(c & 2), altKey: !!(c & 4), shiftKey: !!(c & 8), bubbles: true, cancelable: true }; var acts = !(init.ctrlKey || init.metaKey || init.altKey); var before = st(); var ev = new KeyboardEvent('keydown', init); document.body.dispatchEvent(ev); var after = st(); var changed = []; for (var i = 0; i < 3; i++) { if (before[i] !== after[i]) { changed.push(i); document.getElementById(btns[i]).click(); } } var want = acts ? String([k[1]]) : ''; var label = JSON.stringify(k[0]) + ' c' + c; if (String(changed) !== want) bad.push(label + ' changed [' + changed + ']'); if (k[0] === ' ' && ev.defaultPrevented !== acts) bad.push(label + ' defaultPrevented ' + ev.defaultPrevented); } }); window.__bad = bad.length ? bad.join('; ') : 'ok'; return true; })()
expect window.__bad == ok
expect !document.getElementById('prompter-scroll').classList.contains('flip')
expect !document.getElementById('prompter-scroll').classList.contains('mirror')
expect document.getElementById('pp-play').textContent.indexOf('Play') !== -1

# --- with Ctrl or Alt held the letters leave the text alone ------------------

key alt KeyF
