# B-roll markers: typing "B-roll:" at the start of a line in a visual cell
# has to turn into a .broll-key chip regardless of what character formatting
# is active, the same as it does in plain text. Issue #18: brollAtLineStart()
# used to demand the caret's text node be a DIRECT child of .cell-editable,
# so a marker typed while bold was active (parent is <strong> or a styled
# <span>) was silently rejected and left as plain bold words with no chip.
#
# Six cases against six scenes: a plain-text control, the bug's own repro
# (bold active, typed fresh), bold with a real Enter making a second line,
# bold with a <br> forced to sit INSIDE the bold run (belt-and-braces against
# whatever the browser's own Enter handling happens to nest today), and two
# negatives that must keep failing: mid-sentence, and inside a link.

# --- 4) same shape, forced: the <br> sits INSIDE the bold element itself ---
# Belt-and-braces on top of case 3, which depends on however the browser's
# own Enter handling happens to nest today. This pins the exact structure
# the bug report describes: the caret's text node's parent is a <strong>,
# and what comes before it in that same <strong> is a <br>. A second,
# sentinel <br> follows for the same reason insertLineBreak() (05-render.js)
# adds one: a lone trailing <br> renders no line of its own, so a caret
# "after" it collapses to sit before it instead — exactly the quirk that
# comment describes. With the sentinel in place the caret genuinely lands
# between the two <br>s, i.e. after the first one, same as a real Enter.
eval (function(){var c=document.querySelector('#rows .row:nth-child(4) .visual-cell .cell-editable');c.innerHTML='<strong>Wide shot of the crowd.<br><br></strong>';var strong=c.querySelector('strong');var r=document.createRange();r.setStart(strong,2);r.collapse(true);var s=getSelection();c.focus();s.removeAllRanges();s.addRange(r);return c.innerHTML;})()

# --- 6) negative: "B-roll:" typed inside a link must NOT become a marker ---
# isInsideTagOrLink() is the guard for this, separate from brollAtLineStart()
# and unaffected by its fix. Real per-character typing at the very FIRST
# character of a link makes Chromium insert the new text BEFORE the <a>
# rather than inside it (a browser link-typing heuristic, not app behaviour,
# confirmed by hand against this exact setup) — which would prove nothing
# about isInsideTagOrLink. So this drives the app's own input handler
# directly: the text is already inside the <a>, the caret sits right after
# the colon as if it had just been typed there, and the same 'input' event
# maybeCaptureBroll listens for is dispatched by hand.
eval (function(){var c=document.querySelector('#rows .row:nth-child(6) .visual-cell .cell-editable');c.innerHTML='<a href="https://example.com">B-roll: nope</a>';var t=c.querySelector('a').firstChild;var idx=t.nodeValue.indexOf(':')+1;var r=document.createRange();r.setStart(t,idx);r.collapse(true);var s=getSelection();c.focus();s.removeAllRanges();s.addRange(r);c.dispatchEvent(new InputEvent('input',{bubbles:true,cancelable:true,inputType:'insertText',data:':'}));return c.innerHTML;})()
sleep 600
expect document.querySelectorAll('#rows .row:nth-child(6) .visual-cell .broll-key').length == 0
