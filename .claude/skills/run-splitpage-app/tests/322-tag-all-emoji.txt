# An edit never splits an emoji (#322; src/core/text/edits.ts): a tag name cut
# at 80 UTF-16 units can end in half an emoji, and the Production panel's
# "Tag all N mentions" then tags a range that ends between the two halves.
# The tag takes the whole emoji, so no half is left in or out of the tag.
# Then a caret or a selection set by code inside a pair: paste, cut, Alt+Enter
# and a paste into a comment box each keep the emoji single and whole.
#
# Every `eval` is one physical line (the driver splits on newlines).

# --- a caret or selection placed by code inside a pair ----------------------
# Chromium keeps a programmatic offset between the two halves. Paste, cut,
# Alt+Enter and a paste into a comment box each act on the whole emoji once.
eval (function(){ var s = JSON.parse(F.buildSplitpage()); s.title = 'Emoji paste'; s.rows = [ { id: 'p1', status: 'todo', visualText: 'a' + GRIN + 'b', audioText: 'Alpha <span class="cmt-anchor" data-cmt="c1">bravo</span>' }, { id: 'p2', status: 'todo', visualText: 'a' + GRIN + 'bc', audioText: '' }, { id: 'p3', status: 'todo', visualText: 'a' + GRIN + 'bc', audioText: '' }, { id: 'p4', status: 'todo', visualText: 'a' + GRIN + 'bc', audioText: '' } ]; s.productionTags = []; s.brollShots = []; s.comments = [ { id: 'c1', rowId: 'p1', field: 'audioText', anchorId: 'c1', quotedText: 'bravo', author: 'Ann', body: 'about bravo', createdAt: '2026-09-01T10:00:00Z', kind: 'anchor', replies: [] } ]; F.openFile(new File([JSON.stringify(s)], 'emoji-paste.splitpage')); return true; })()
wait F.state().title === 'Emoji paste' && document.querySelector('#rows .row[data-id="p4"]')

eval (function(){ window.vis = function (id) { return document.querySelector('#rows .row[data-id="' + id + '"] .visual-cell .cell-editable'); }; window.st = function (id) { return F.state().rows.filter(function (r) { return r.id === id; })[0].visualText; }; window.place = function (el, from, to) { el.focus(); var w = document.createTreeWalker(el, NodeFilter.SHOW_TEXT), n, seen = 0, a = null, b = null; while ((n = w.nextNode())) { var len = n.nodeValue.length; if (!a && from <= seen + len) a = [n, from - seen]; if (!b && to <= seen + len) b = [n, to - seen]; seen += len; } var r = document.createRange(); r.setStart(a[0], a[1]); r.setEnd(b[0], b[1]); var s = getSelection(); s.removeAllRanges(); s.addRange(r); return s.getRangeAt(0).startOffset; }; window.clip = function (type, el, data) { var dt = new DataTransfer(); if (data) Object.keys(data).forEach(function (k) { dt.setData(k, data[k]); }); var ev = new ClipboardEvent(type, { clipboardData: dt, bubbles: true, cancelable: true }); el.dispatchEvent(ev); return { prevented: ev.defaultPrevented, text: dt.getData('text/plain') }; }; window.before = function (el) { var s = getSelection(); var r = document.createRange(); r.selectNodeContents(el); r.setEnd(s.anchorNode, s.anchorOffset); return r.toString().replace(/​/g, ''); }; return true; })()

# a paste over a selection that starts inside the pair replaces the whole emoji
