# Opening a foreign document starts a NEW document (issue #7).
#
# loadFileIntoEditor (src/js/18-local-files.js) used to replace only
# state.rows and the title, keeping productionTags, people, brollShots and
# comments — and keeping the Save target — so an unrelated .docx/.md/.txt/.csv
# inherited the previous script's production breakdown, cast list and review
# comments, and Save still wrote to the previous project's file.
#
# Decision: File ▸ Open of a foreign file is a new document. The "revised
# draft of the same script" workflow it used to serve is File ▸ Paste a
# script…, which is unchanged in what it keeps and now takes a picked file
# as well as pasted text. Both halves are asserted here.

open

eval (function(){window.F = window.__splitpageFormats; return !!F;})()
expect true

# --- a project with every list populated, adopted from a .splitpage file ---
# The seed has one comment and nothing else, so the starting document is
# built here: a tag, a person owning it, a b-roll shot and the seed comment.
eval (function(){ var s = F.clone(F.state()); s.people = [{id:'p1', name:'Ada'}]; s.productionTags = [{cat:'gear', name:'Tripod', status:'todo', owner:'p1', notes:'from the old script'}]; s.brollShots = [{id:'bs1', name:'City skyline', origin:'list'}]; window.__seeded = F.buildSplitpage(s); return window.__seeded.length > 0; })()
eval (async function(){ await F.openFile(new File([window.__seeded], 'old-project.splitpage')); return true; })()
wait F.docFile().kind === 'splitpage'
expect F.state().productionTags.length == 1
expect F.state().people.length == 1
expect F.state().brollShots.length == 1
expect F.state().comments.length == 1
eval (function(){ window.__oldId = F.state().docId; window.__oldRows = F.state().rows.length; return window.__oldId; })()
expect /^d[0-9a-f]{32}$/.test(window.__oldId) == true

# --- opening a plain .txt leaves none of it behind ------------------------

expect F.state().productionTags.length == 0
expect F.state().people.length == 0
expect F.state().brollShots.length == 0
expect F.state().comments.length == 0
# and neither home of the Save button still points at the old project

# the new id has never left this page, so a first Save can take it with it

# the panels agree with the document: no badges carried over

expect document.getElementById('broll-badge').hidden == true
expect document.getElementById('comments-badge').hidden == true
# the chip names the foreign file it is round-tripping, not the old project
expect document.getElementById('file-chip-name').textContent == stranger.txt

# --- what each foreign format's rows hold after File ▸ Open -----------------
# Every reader behind loadFileIntoEditor hands back HTML already: a .txt or a
# foreign .md has been through legacyMdToHtml once in normalizeImportedRows,
# a .csv is escaped cell by cell, a .docx is built as markup (41-export-
# polish.txt). The new document used to be built by migrateRows() with no
# schema, which ran legacyMdToHtml over that HTML a second time: "&" became
# a visible "&amp;", and Markdown's **bold** a visible "<strong>".
# __frag parses a stored cell once, the way the editor renders it.
eval (function(){ window.__frag = function(h){ var t = document.createElement('template'); t.innerHTML = h || ''; return t.content; }; return true; })()

# a .txt: "&" and "<" are text, and stay one level of escaping deep
eval (async function(){ await F.openFile(new File(['Tom & Jerry <cut> here.' + String.fromCharCode(10,10) + 'Second & last.'], 'ampersand.txt')); return F.state().rows.map(function(r){ return r.audioText; }).join(' | '); })()
wait F.state().title === 'ampersand'
expect F.state().rows.length == 2
expect __frag(F.state().rows[0].audioText).textContent === 'Tom & Jerry <cut> here.'
expect __frag(F.state().rows[0].audioText).querySelector('*') === null
expect F.state().rows[0].audioText === 'Tom &amp; Jerry &lt;cut&gt; here.'
expect document.querySelector('#rows .row .audio-cell .cell-editable').textContent === 'Tom & Jerry <cut> here.'

# a foreign .md in prose: its Markdown becomes real formatting, once
eval (async function(){ await F.openFile(new File(['# Marked Up' + String.fromCharCode(10,10) + 'Some **bold words** and *an aside* & [a link](https://example.com/x?a=1&b=2).'], 'marked.md')); return F.state().rows.map(function(r){ return r.audioText; }).join(' | '); })()
wait F.state().title === 'Marked Up'
eval (function(){ window.__mdRow = F.state().rows.filter(function(r){ return r.audioText.indexOf('bold') !== -1; })[0]; return window.__mdRow ? window.__mdRow.audioText : null; })()
expect __mdRow.audioText.indexOf('<strong>bold words</strong>') !== -1
expect __mdRow.audioText.indexOf('<em>an aside</em>') !== -1
expect (function(){ var a = __frag(__mdRow.audioText).querySelector('a'); return !!a && a.getAttribute('href') === 'https://example.com/x?a=1&b=2'; })()
expect __frag(__mdRow.audioText).textContent === 'Some bold words and an aside & a link.'

# a foreign .md table: both columns, their Markdown formatted, once
eval (async function(){ var nl = String.fromCharCode(10); await F.openFile(new File(['| Visual | Audio |' + nl + '| --- | --- |' + nl + '| **Wide** & far | Say *hello* <now> |' + nl + '| Close up | Plain & simple |' + nl], 'table.md')); return F.state().rows.map(function(r){ return r.visualText + ' / ' + r.audioText; }).join(' | '); })()
wait F.state().title === 'table'
expect F.state().rows.length == 2
expect F.state().rows[0].visualText.indexOf('<strong>Wide</strong>') !== -1
expect __frag(F.state().rows[0].visualText).textContent === 'Wide & far'
expect F.state().rows[0].audioText.indexOf('<em>hello</em>') !== -1
expect __frag(F.state().rows[0].audioText).textContent === 'Say hello <now>'
expect __frag(F.state().rows[1].audioText).textContent === 'Plain & simple'

# a .csv: every cell is text, markup-looking or not
