# B-roll anchoring (issue #21): a "B-roll:" line is a marker however it got
# into the script, not only when it was typed. maybeCaptureBroll() is the
# only writer of a .broll-key span and it fires on a live keystroke, so an
# imported or pasted line counted on the badge but carried no id — and an
# id is the only thing that survives a rewording. This drives the two
# routes the issue measured (import, paste) and then the consequence that
# made it matter: a status set on an imported line, then the line reworded.

# --- 2) paste: the same line arriving through the clipboard ---------------
# A real paste event through the app's own onRootPaste, not a programmatic
# innerHTML set, so this is the route the issue measured. While the caret is
# still in the field nothing is rewritten — anchoring never fights a live
# edit — and leaving the field is what anchors it.
#
# Leaving it has to be a REAL click, not activeElement.blur(): an unfocused
# headless window moves activeElement without firing focus/blur at all, so
# blur() leaves the app's focusout handler (and every sanitiser hanging off
# it) unrun. Same reason the skill's gotchas list `jsclick` as not blurring.
caret #rows .row:nth-child(1) .visual-cell .cell-editable
eval (function(){var c=document.querySelector('#rows .row:nth-child(1) .visual-cell .cell-editable');var r=document.createRange();r.selectNodeContents(c);var s=getSelection();c.focus();s.removeAllRanges();s.addRange(r);document.execCommand('delete');var dt=new DataTransfer();dt.setData('text/plain','B-roll: pasted puffins');c.dispatchEvent(new ClipboardEvent('paste',{bubbles:true,cancelable:true,clipboardData:dt}));return c.textContent;})()
sleep 700
expect document.querySelector('#rows .row:nth-child(1) .visual-cell .cell-editable').textContent.indexOf('pasted puffins') !== -1
expect document.querySelectorAll('#rows .row:nth-child(1) .visual-cell .broll-key').length == 0
