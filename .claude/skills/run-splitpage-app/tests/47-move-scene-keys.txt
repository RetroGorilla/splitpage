# Alt+Up / Alt+Down move the scene the caret is in one place earlier or
# later (issue #24): swapAdjacentRows() in 06-rows.js (undone since #85 by the document-wide undo, 12a-undo.js), reusing
# the exact record-and-sync path moveRowTo() (drag-reorder) already uses.
# Covers both #rows and the writer, no wrap at either end, one Ctrl+Z undo
# step, tags/status/comments travelling with the scene, the production
# breakdown picking up the new order, and never firing inside a comment
# reply box. Selectors are id-based throughout rather than nth-child, since
# position is exactly what this feature changes.

open

# capture the seed rows' own ids up front
eval (function(){var rows=document.querySelectorAll('#rows .row');window.__r1=rows[0].getAttribute('data-id');window.__r2=rows[1].getAttribute('data-id');return [window.__r1,window.__r2];})()

# --- set up: a third scene, and give the seed's SECOND scene (which already
# carries a comment, seeded anchored to "word for word") a tag and a status
# too, so a move can be checked as a whole -----------------------------------
eval (function(){document.querySelector('.add-row-btn').click();return true;})()
sleep 300
expect document.querySelectorAll('#rows .row').length == 3
eval (function(){window.__r3=document.querySelectorAll('#rows .row')[2].getAttribute('data-id');return window.__r3;})()

eval (function(){var cell=document.querySelector('#rows .row[data-id="'+window.__r2+'"] .visual-cell .cell-editable');cell.focus();var r=document.createRange();r.setStart(cell.firstChild,0);r.collapse(true);var s=getSelection();s.removeAllRanges();s.addRange(r);return true;})()
press  [gaffer tape]
sleep 400
expect !!document.querySelector('#rows .row[data-id="'+window.__r2+'"] .tag-anchor[data-tag-name="gaffer tape"]')

eval document.activeElement.blur()
eval (function(){document.querySelector('#rows .row[data-id="'+window.__r2+'"] .status-btn').click();document.querySelector('#rows .row[data-id="'+window.__r2+'"] .status-btn').click();return true;})()
sleep 400
expect document.querySelector('#rows .row[data-id="'+window.__r2+'"]').classList.contains('is-shot')

# --- Alt+Up on the FIRST scene: no wrap, nothing happens --------------------
eval (function(){var cell=document.querySelector('#rows .row[data-id="'+window.__r1+'"] .audio-cell .cell-editable');cell.focus();var t=cell.firstChild;var i=t.nodeValue.indexOf('welcome');var r=document.createRange();r.setStart(t,i);r.collapse(true);var s=getSelection();s.removeAllRanges();s.addRange(r);return i;})()

sleep 400

expect document.activeElement.getAttribute('data-field') == audioText

# --- Alt+Down on the LAST scene: no wrap, nothing happens -------------------
eval (function(){var cell=document.querySelector('#rows .row[data-id="'+window.__r3+'"] .audio-cell .cell-editable');cell.focus();var r=document.createRange();r.selectNodeContents(cell);r.collapse(false);var s=getSelection();s.removeAllRanges();s.addRange(r);return true;})()
key alt ArrowDown
sleep 400

expect document.activeElement.closest('.row').getAttribute('data-id') === window.__r3

# --- Alt+Up moves the tagged/statused/commented middle scene up a place,
# taking everything with it, and the caret lands back at the same field and
# the same plain-text offset -------------------------------------------------
eval (function(){var cell=document.querySelector('#rows .row[data-id="'+window.__r2+'"] .audio-cell .cell-editable');cell.focus();var t=cell.firstChild;var i=t.nodeValue.indexOf('left');var r=document.createRange();r.setStart(t,i);r.collapse(true);var s=getSelection();s.removeAllRanges();s.addRange(r);window.__r2Offset=i;return i;})()
key alt ArrowUp
sleep 400
expect document.querySelectorAll('#rows .row')[0].getAttribute('data-id') === window.__r2

# the tag, the status and the comment all moved with it
