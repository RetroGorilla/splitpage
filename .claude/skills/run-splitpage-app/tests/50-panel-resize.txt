# Every side panel resizable, not just Comments (issue #50, 03b-side-panels.js
# + 03-sync.js's openSidePanel). Each of Comments/Production/Gear/B-roll/
# History gets its own drag handle and remembers its own width in one JSON
# map (splitpage-panel-widths), with the pre-existing
# splitpage-comments-panel-width honoured once as Comments' own initial
# value until it is actually dragged.
#
# Drag is driven with synthetic mousedown/mousemove/mouseup on the handle and
# on document, the same way 33-drag-autoscroll.txt drives a scene drag —
# there is no real pointer session to produce over CDP. Each handle's
# mousedown listener adds document-level mousemove/mouseup, so the synthetic
# events are dispatched there rather than on the handle itself.
#
# Every `eval` here has to be one line — the driver reads the script line by
# line, so a JS statement broken across lines is read as separate unknown
# commands.
#
# viewport is set once at the top (>=1100px, where #root's padding-right
# tracks the open panel) and never changed again — see the SKILL's gotcha on
# changing device metrics mid-suite aborting the CDP socket. The goto reload
# for the persistence check is kept at the very end, with only one small
# write right before it, per the same SKILL's note on a busy suite aborting
# on its own trailing goto.

open
viewport 1400 900
goto /index.html
sleep 500

eval (function(){window.__panelWidth=function(id){return parseFloat(getComputedStyle(document.getElementById(id)).getPropertyValue('--panel-width'));};window.__rootPadding=function(){return parseFloat(getComputedStyle(document.getElementById('root')).paddingRight);};window.__dragHandle=function(handleId,startX,endX){var handle=document.getElementById(handleId);var down=new MouseEvent('mousedown',{bubbles:true,cancelable:true,clientX:startX,clientY:220});handle.dispatchEvent(down);document.dispatchEvent(new MouseEvent('mousemove',{bubbles:true,cancelable:true,clientX:endX,clientY:220}));document.dispatchEvent(new MouseEvent('mouseup',{bubbles:true,cancelable:true,clientX:endX,clientY:220}));return down.defaultPrevented;};return true;})()
expect true

# =============================================================================
# legacy splitpage-comments-panel-width is honoured as Comments' OWN initial
# width the first time it opens after the change — before anything else has
# ever written the new splitpage-panel-widths map.
# =============================================================================
expect localStorage.getItem('splitpage-panel-widths') === null
eval (function(){localStorage.setItem('splitpage-comments-panel-width', '333'); return true;})()
jsclick #btn-comments
sleep 300
expect !document.getElementById('comments-panel').hidden
expect window.__panelWidth('comments-panel') === 333

# =============================================================================
# defaults differ per panel: Production and B-roll start wider (520) than
# Comments/Gear/History (400) — same clamp range (280-720) for all five.
# =============================================================================
jsclick #btn-production
sleep 300
expect window.__panelWidth('production-panel') === 520
jsclick #btn-gear
sleep 300
expect window.__panelWidth('gear-panel') === 400
jsclick #btn-broll
sleep 300
expect window.__panelWidth('broll-panel') === 520
jsclick #btn-file
jsclick #file-menu [data-file-action="history"]
sleep 300
expect window.__panelWidth('history-panel') === 400

# =============================================================================
# drag never starts a text selection or a scene drag underneath — the
# handle's mousedown calls preventDefault()
# =============================================================================
jsclick #btn-production
sleep 300
expect window.__dragHandle('production-panel-resize-handle', 500, 470) === true

# =============================================================================
# clamp: drag far past 720 on one side, far past 280 on the other
# =============================================================================
eval window.__dragHandle('production-panel-resize-handle', 600, -600)
expect window.__panelWidth('production-panel') === 720
eval window.__dragHandle('production-panel-resize-handle', 300, 1200)
expect window.__panelWidth('production-panel') === 280

# =============================================================================
# double-click resets to that panel's own default
# =============================================================================
eval (function(){document.getElementById('production-panel-resize-handle').dispatchEvent(new MouseEvent('dblclick', {bubbles:true, cancelable:true}));return true;})()
expect window.__panelWidth('production-panel') === 520

# =============================================================================
# per-panel independence: widen Production, open Comments — Comments is
# unaffected — then reopen Production and it is still the wide one.
# =============================================================================
eval window.__dragHandle('production-panel-resize-handle', 500, 370)
expect window.__panelWidth('production-panel') === 650
jsclick #btn-comments
sleep 300
expect window.__panelWidth('comments-panel') === 333
expect document.getElementById('production-panel').hidden
jsclick #btn-production
sleep 300
expect window.__panelWidth('production-panel') === 650

# =============================================================================
# #root's padding-right (>=1100px) tracks whichever panel is actually open,
# for every panel — not a fixed guess
# =============================================================================
expect window.__rootPadding() === 670
jsclick #btn-comments
sleep 300
expect window.__rootPadding() === 353
jsclick #btn-gear
sleep 300
expect window.__rootPadding() === 420
jsclick #btn-broll
sleep 300
expect window.__rootPadding() === 540
jsclick #btn-file
jsclick #file-menu [data-file-action="history"]
sleep 300
expect window.__rootPadding() === 420

# =============================================================================
# persistence across reload — kept minimal and at the very end. Narrow Gear
# a little first (still comfortably inside the clamp), then reload and check
# Gear, Production and Comments all come back the width they were left at —
# Comments still by way of the legacy key, since it has never actually been
# dragged in this suite.
# =============================================================================
jsclick #btn-gear
sleep 300
eval window.__dragHandle('gear-panel-resize-handle', 500, 600)
expect window.__panelWidth('gear-panel') === 300

goto /index.html
sleep 900
eval (function(){window.__panelWidth=function(id){return parseFloat(getComputedStyle(document.getElementById(id)).getPropertyValue('--panel-width'));};return true;})()
jsclick #btn-gear
sleep 300
expect window.__panelWidth('gear-panel') === 300
jsclick #btn-production
sleep 300
expect window.__panelWidth('production-panel') === 650
jsclick #btn-comments
sleep 300
expect window.__panelWidth('comments-panel') === 333

quit
