# The compare-and-set save and its 409 merge (saveToServer, 13-hosted.js).
#
# This is the merge the README's collaboration story rests on and the only
# one of the three that had no suite at all: the harness serves static files
# and cannot start server.js, so there was no 409 to be had. There is one
# here because `fetch` is replaced for PUTs to api/docs/ only — a queue of
# answers in, a record of exactly what was sent out — which is enough to
# drive the whole loop and to read back the two numbers that decide whether
# anybody's work is lost. Everything else (GET, presence, update-status)
# still goes to the real static server.
#
# The rest of the session comes from window.__hostedTest.setSession(), the
# same hook 43-review-mode.txt uses.
#
# What it pins down (#72):
#   * baseRev is the DOCUMENT's rev, taken from the server's own answers and
#     nowhere else — 3 from the session, then 4 out of the 409 body, then 5
#     out of the 200 body;
#   * `state.rev` is NOT pre-incremented by the client. It is the merge
#     counter inside the document (max-plus-one, 02-state.js) and moves only
#     when a merge moves it. The client used to bump it on the way out and
#     put it back on every failure, which was a second, private counter that
#     drifted from the server's on every conflict;
#   * both sides' edits survive the merge, and the banner only appears once
#     the merged version is actually stored (reportMerge's alreadySaved).

# The fake server: PUTs to api/docs/ are recorded and answered from
# __answers; everything else goes through to the real one.
eval (function(){ window.__puts = []; window.__answers = []; var real = window.fetch.bind(window); window.fetch = function(url, opts){ var u = String(url); if (opts && opts.method === 'PUT' && u.indexOf('api/docs/') !== -1) { window.__puts.push(JSON.parse(opts.body)); var a = window.__answers.shift(); return Promise.resolve(new Response(JSON.stringify(a.body), {status: a.status, headers: {'content-type':'application/json'}})); } return real(url, opts); }; return true; })()
