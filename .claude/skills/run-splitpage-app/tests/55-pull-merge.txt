# The pull channel: a rev on the presence heartbeat, and a merge when it
# moves (notePresenceRev / pullServerDoc, 13-hosted.js — issue #69).
#
# TWO CLIENTS, one page. This page is client B. Client A is the document
# sitting on the fake server: `window.__server` is what A last saved, and
# window.__serverRev is the rev A left behind. Nothing else about A needs
# simulating — a second editor is only ever a rev this view has not seen.
#
# `fetch` is replaced the same way tests/54-server-conflict.txt replaces it,
# and for the same reason: the harness serves static files and cannot start
# server.js. Here it answers three things and passes everything else through
# to the real static server — POST api/presence (the heartbeat, which is the
# whole channel), GET api/docs/doc-pull (what the pull fetches), and PUT
# api/docs/ (recorded, and held open on purpose, so the suite can ask what
# happens to a pull that lands in the middle of a save).
#
# What it pins down:
#   * B sees A's row within 3s of A saving, without B saving anything —
#     driven by the REAL two-second heartbeat, not by poking the function;
#   * B's own unsaved edit in another row survives it;
#   * no PUT: a pull publishes nothing, not even the merged result;
#   * the rev compared is the DOCUMENT's (serverDoc.rev, the compare-and-set
#     token), so a beat that repeats a number costs one comparison and no
#     request at all;
#   * NO HISTORY SNAPSHOT. A pull runs whenever anybody saves; one undo
#     point per pull would fill the fifty-entry history in under two minutes
#     and evict the versions the user actually wants;
#   * a rev that moved but brought nothing new is silent — no toast, no
#     re-render, no dirty mark;
#   * and nothing is pulled while a save is in flight, because the merge
#     would be against a baseState that save is about to replace.

# number a GET comes back at; they are one value on purpose, because on a
# real server they are one number in one file.
eval (function(){ window.__gets = 0; window.__beats = 0; window.__puts = []; window.__putResolve = null; window.__serverRev = 3; window.__server = null; window.__serverBy = 'Priya Malhotra'; var real = window.fetch.bind(window); window.fetch = function(url, opts){ var u = String(url); opts = opts || {}; if (opts.method === 'POST' && u.indexOf('api/presence') !== -1) { window.__beats++; return Promise.resolve(new Response(JSON.stringify({docId:'doc-pull', rev: window.__serverRev, here: []}), {status:200, headers:{'content-type':'application/json'}})); } if (opts.method === 'PUT' && u.indexOf('api/docs/') !== -1) { window.__puts.push(JSON.parse(opts.body)); return new Promise(function(done){ window.__putResolve = function(body){ window.__putResolve = null; done(new Response(JSON.stringify(body), {status:200, headers:{'content-type':'application/json'}})); }; }); } if (u.indexOf('api/docs/doc-pull') !== -1) { window.__gets++; return Promise.resolve(new Response(JSON.stringify({id:'doc-pull', rev: window.__serverRev, state: window.__server, updatedAt: new Date().toISOString(), updatedBy: window.__serverBy}), {status:200, headers:{'content-type':'application/json'}})); } return real(url, opts); }; return true; })()

# --- a rev that moved but brought nothing is silent ----------------------
# A save that changed no words — their own pull-merge catching up, a save
# that only moved `lastEditedAt`, a second window of theirs. `__server` is
# left exactly as it was and only the rev moves, which is what that looks
# like from here. The rev is taken so the next beat is quiet; nothing else
# happens at all. Note what is NOT the test: "the same as what is on my
# screen". B's BROWONE is still unsaved, so the screen and the server have
# differed all along; the question is whether anything of THEIRS is new,
