# A save to a server document that has been deleted (saveToServer's
# not-found branch, 13-hosted.js, #99).
#
# The branch had never run: apiFetch() named every non-ok status by its
# number, so the server's 404 {error:"not-found"} arrived as code "404" and
# `err.code === "not-found"` was always false — a save to a deleted document
# fell through to handleApiError()'s generic "Could not save to the server —
# not-found." error toast and left the page still calling the server its
# home, so every later Save and autosave PUT at a document that was not
# there. apiFetch() now names the server's own "no such document" 404
# "not-found", and this suite is what would have caught it.
#
# `fetch` is replaced for PUTs to api/docs/ only, the way 54-server-conflict
# replaces it; the rest of the session is window.__hostedTest.setSession().
#
# What it pins down:
#   * a 404 {error:"not-found"} for the PUT runs the not-found path: the
#     warning that names it, and the server home cleared through setHome()
#     (#73) to "none" — not the generic error toast;
#   * what was on screen stays, and stays dirty: it is the only copy now;
#   * the same path runs when the document goes away between a 409 and the
#     retry the merge makes;
#   * a 404 about something else (not the server's "not-found") is still
#     named by its number and still gets the generic handling.

# The fake server: PUTs to api/docs/ are recorded and answered from
# __answers; everything else goes through to the real one.
eval (function(){ window.__puts = []; window.__answers = []; var real = window.fetch.bind(window); window.fetch = function(url, opts){ var u = String(url); if (opts && opts.method === 'PUT' && u.indexOf('api/docs/') !== -1) { window.__puts.push(JSON.parse(opts.body)); var a = window.__answers.shift(); return Promise.resolve(new Response(JSON.stringify(a.body), {status: a.status, headers: {'content-type':'application/json'}})); } return real(url, opts); }; return true; })()
expect true

# Every toast, as it is put up — they fade on their own, so reading the host
