# The image blob registry and its IndexedDB backend (src/js/02c-blobs.js,
# issue #82 P1). Nothing in the app calls the registry yet, so everything is
# driven through window.__blobsTest: real encoder output from a canvas
# (PNG, JPEG, WebP) measured and hashed, the data: URI decoder held to what
# this browser itself decodes, then intern -> persist -> reload -> ensureBytes
# against the real 'splitpage-blobs' database, and the bytes shown again
# through a blob: URL under the page's own CSP.
#
# The pure halves have unit tests (src/test/blob-policy.test.mjs,
# blob-store.test.mjs); what only a browser can say is here. Every `eval`
# is one physical line.

# ---- helpers: canvas images, the browser's own verdict on a data: URI ---
eval (function(){ window.B = window.__blobsTest; window.canvasUrl = function(type){ var c = document.createElement('canvas'); c.width = 37; c.height = 23; var g = c.getContext('2d'); g.fillStyle = '#c33'; g.fillRect(0, 0, 37, 23); g.fillStyle = '#39f'; g.fillRect(5, 4, 20, 9); return c.toDataURL(type, 0.9); }; window.loads = function(src){ return new Promise(function(r){ var i = new Image(); i.onload = function(){ r(i.naturalWidth + 'x' + i.naturalHeight); }; i.onerror = function(){ r(null); }; i.src = src; }); }; window.b64bytes = function(u){ var bin = atob(u.slice(u.indexOf(',') + 1)); var a = new Uint8Array(bin.length); for (var k = 0; k < bin.length; k++) a[k] = bin.charCodeAt(k); return a; }; window.subtleHex = async function(bytes){ var d = new Uint8Array(await crypto.subtle.digest('SHA-256', bytes)); return Array.prototype.map.call(d, function(x){ return ('0' + x.toString(16)).slice(-2); }).join(''); }; return true; })()

# ---- real encoder output: type sniffed, size measured, sha = subtle's ----
eval (async function(){ window.recs = {}; for (var t of ['image/png', 'image/jpeg', 'image/webp']) { var u = canvasUrl(t); window.recs[t] = { url: u, rec: B.internDataUrl(u), subtle: await subtleHex(b64bytes(u)) }; } return JSON.stringify(Object.keys(recs).map(function(k){ return recs[k].rec; })); })()

expect recs['image/webp'].rec.mime === 'image/webp' && recs['image/webp'].rec.width === 37 && recs['image/webp'].rec.height === 23
expect recs['image/png'].rec.sha256 === recs['image/png'].subtle && recs['image/jpeg'].rec.sha256 === recs['image/jpeg'].subtle && recs['image/webp'].rec.sha256 === recs['image/webp'].subtle
# the canonical inline form is what the canvas produced, byte for byte
expect B.dataUrlForSync(recs['image/png'].rec.sha256) === recs['image/png'].url

# ---- the decoder refuses exactly what this browser refuses to show ------
# Each variant of one PNG: does internDataUrl give the original sha, and does
# an <img> load it? The two answers must agree for every variant.
eval (async function(){ var u = recs['image/png'].url, head = u.slice(0, u.indexOf(',') + 1), body = u.slice(head.length), sha = recs['image/png'].rec.sha256; var vs = { wrapped: head + body.replace(/(.{60})/g, '$1\n'), crlf: head + body.replace(/(.{40})/g, '$1\r\n'), spaced: head + body.replace(/(.{4})/g, '$1 '), tabs: head + '\t' + body + '\t', formfeed: head + body.slice(0, 16) + '\f' + body.slice(16), unpadded: head + body.replace(/=+$/, ''), trailingVT: head + body + '\u000b', jpgLabel: 'data:image/jpg;base64,' + body, innerNBSP: head + body.slice(0, 16) + '\u00a0' + body.slice(16), innerVT: head + body.slice(0, 16) + '\u000b' + body.slice(16), midPad: head + body.slice(0, 16) + '=' + body.slice(16) }; window.verdicts = {}; for (var k in vs) { var r = B.internDataUrl(vs[k]); window.verdicts[k] = { ours: r ? r.sha256 === sha : null, browser: (await loads(vs[k])) === '37x23' }; } return JSON.stringify(verdicts); })()
expect Object.keys(verdicts).every(function(k){ return (verdicts[k].ours === true) === verdicts[k].browser; })
expect Object.keys(verdicts).every(function(k){ return verdicts[k].ours !== false; })
expect verdicts.wrapped.browser && verdicts.unpadded.browser && verdicts.trailingVT.browser
expect !verdicts.innerNBSP.browser && !verdicts.innerVT.browser && !verdicts.midPad.browser
