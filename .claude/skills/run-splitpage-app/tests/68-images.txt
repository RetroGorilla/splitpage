# Images out of the document state (#82 P2b): a picture is `row.imageRef`,
# a record naming bytes in the image registry (02c-blobs.js), and the
# bytes live in the 'splitpage-blobs' IndexedDB database, not in `state`.
# Three pictures dropped onto three scenes the way a user drops them (a
# real DragEvent carrying a File) leave the state under 10 KB and show
# through blob: URLs; the crash/sign-in stash carries them across a real
# reload as refs and they render again from IndexedDB; an old document
# whose pictures are base64 data: URIs opens content-addressed (its
# imageRef is the SHA-256 of the decoded bytes) and still shows; and a
# .splitpage save carries every picture inline, as the format always has.
# Every `eval` is one physical line.

# ---- helpers: a noisy canvas PNG (so it is a real ~100 KB picture), a
# File from it, a drop of that File onto a scene's visual column, and
# crypto.subtle as the independent oracle for the sha
eval (function(){ window.F = window.__splitpageFormats; window.B = window.__blobsTest; window.pngUrl = function(w, h, seed){ var c = document.createElement('canvas'); c.width = w; c.height = h; var g = c.getContext('2d'); var d = g.createImageData(w, h); var x = seed * 2654435761 >>> 0; for (var i = 0; i < d.data.length; i++) { x = (x * 1103515245 + 12345) >>> 0; d.data[i] = (i % 4 === 3) ? 255 : (x >>> 16) & 255; } g.putImageData(d, 0, 0); return c.toDataURL('image/png'); }; window.bytesOf = function(u){ var bin = atob(u.slice(u.indexOf(',') + 1)); var b = new Uint8Array(bin.length); for (var i = 0; i < bin.length; i++) b[i] = bin.charCodeAt(i); return b; }; window.subtleHex = async function(bytes){ var d = new Uint8Array(await crypto.subtle.digest('SHA-256', bytes)); return Array.prototype.map.call(d, function(x){ return ('0' + x.toString(16)).slice(-2); }).join(''); }; window.dropOn = function(rowId, url, name){ var dt = new DataTransfer(); dt.items.add(new File([bytesOf(url)], name, { type: 'image/png' })); var target = document.querySelector('#rows .row[data-id="' + rowId + '"] .visual-media'); target.dispatchEvent(new DragEvent('drop', { bubbles: true, cancelable: true, dataTransfer: dt })); return true; }; window.imgOf = function(rowId){ return document.querySelector('#rows .row[data-id="' + rowId + '"] .visual-media img'); }; return true; })()
