# Tags and b-roll keys made through the document model (#85, design §5).
#
# The typed routes to a tag — a closing shortcut, Tab on the completion
# ghost, Tab on an untagged mention — and typing "B-roll:" used to put their
# span in with execCommand("insertHTML"), which only the browser's own undo
# stack knew about. They are edits of the cell's model now (editCell,
# 05b-cell-edit.js): each is ONE undo entry of its own, after the typing's,
# so one Ctrl+Z gives back exactly what was typed, and the caret lands just
# after the new tag or key, outside it.
#
# Every `eval` is one physical line (the driver splits on newlines).

open
wait document.querySelectorAll('#rows .row').length > 1
eval (function(){ window.U = window.__undoTest; window.F = window.__splitpageFormats; window.cell = function (i, col) { return document.querySelectorAll('#rows .row')[i].querySelector('.' + col + '-cell .cell-editable'); }; window.st = function (i, col) { return F.state().rows[i][col + 'Text']; }; window.reg = function (name) { return F.state().productionTags.some(function (t) { return t.name === name; }); }; return true; })()

# --- a typed [shortcut]: one entry, and Ctrl+Z gives back the typed text -----
caret #rows .row:nth-child(1) .audio-cell .cell-editable
press  a [zebra lamp
sleep 200
eval (function(){ U.flush(); window.__n = U.sizes().undo; return window.__n; })()
press ]
sleep 300
expect cell(0, 'audio').querySelectorAll('.tag-anchor[data-tag-name="zebra lamp"]').length == 1
expect cell(0, 'audio').querySelector('.tag-anchor[data-tag-name="zebra lamp"]').getAttribute('data-tag-cat') == prop
expect st(0, 'audio').indexOf('[zebra lamp]') === -1
expect reg('zebra lamp')
# the "]" joined the word it closed; the tag is the one entry after it
expect U.sizes().undo === window.__n + 1
# the caret is just after the tag, outside it — so typing carries on as plain text
expect (function(){ var s = getSelection(); var n = s.anchorNode; var el = n.nodeType === 3 ? n.parentNode : n; return s.isCollapsed && !el.closest('.tag-anchor') && cell(0, 'audio').contains(n); })()
press  on the desk
sleep 300
expect cell(0, 'audio').querySelector('.tag-anchor[data-tag-name="zebra lamp"]').textContent == zebra lamp
expect /zebra lamp<\/span>(\s|&nbsp;)on the desk$/.test(st(0, 'audio'))
# undo the typing after it, then the tag: the typed "[zebra lamp]" is back,
# and the registry entry the tag made goes with it
eval (function(){ var n = 0; while (U.sizes().undo > window.__n + 1 && n++ < 20) { var c = cell(0, 'audio'); c.dispatchEvent(new InputEvent('beforeinput', { inputType: 'historyUndo', bubbles: true, cancelable: true })); } return U.sizes().undo; })()
expect st(0, 'audio').indexOf('on the desk') === -1
expect cell(0, 'audio').querySelectorAll('.tag-anchor[data-tag-name="zebra lamp"]').length == 1
key ctrl KeyZ
sleep 300
expect cell(0, 'audio').querySelectorAll('.tag-anchor[data-tag-name="zebra lamp"]').length == 0
expect /a \[zebra lamp\]$/.test(st(0, 'audio').replace(/&nbsp;/g, ' '))
expect !reg('zebra lamp')
# and redo puts the tag back, with its registry entry
key ctrl+shift KeyZ
sleep 300
expect cell(0, 'audio').querySelectorAll('.tag-anchor[data-tag-name="zebra lamp"]').length == 1
expect reg('zebra lamp')

# --- a b-roll key: typed, then undone ---------------------------------------
caret #rows .row:nth-child(1) .visual-cell .cell-editable

key ctrl KeyZ

# No redo check here: the undo's own repaint reads the b-roll list, and that
# read (brollEntries) makes a record for the plain "B-roll:" line it now
# finds — a document change of its own, which empties the redo stack. That
# is the list's reader writing, not this path; the tag checks above cover
# redo.
