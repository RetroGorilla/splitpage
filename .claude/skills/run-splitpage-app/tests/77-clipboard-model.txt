# Copy, cut, paste and drop in a script cell through the document model
# (#85, design §5, D6, D7; src/js/05c-clipboard.js).
#
#   * a paste of rich HTML lands formatted, sanitised, with the caret after
#     it, as ONE undo entry;
#   * a copy of commented words never carries the comment: pasted
#     elsewhere, the words come and the comment stays where it was;
#   * a cut then a paste in this tab MOVES the comment: its words carry the
#     anchor, and the comment's row/column follow them, so its card says
#     the new place and jumps there; a second paste of the same cut, or
#     another tab's cut, brings no comment;
#   * a picture or a video card pasted or dropped into a script cell is
#     not put in it — since #135 it goes into the scene's gallery (its
#     Image section), which tests/92-paste-into-gallery.txt covers in full;
#   * a drop goes in through the model too.
# The pure half is src/test/clipboard.test.mjs. Every `eval` is one
# physical line.

eval (function(){ window.cell = function(id, field){ return document.querySelector('#rows .row[data-id="' + id + '"] .' + (field === 'visualText' ? 'visual' : 'audio') + '-cell .cell-editable'); }; window.st = function(id, field){ return F.state().rows.filter(function(r){ return r.id === id; })[0][field]; }; window.selectText = function(el, needle, collapseEnd){ el.focus(); var text = el.textContent; var at = text.indexOf(needle); if (at === -1) throw new Error('no ' + needle); var from = at, to = at + needle.length; if (collapseEnd === 'end') { from = to = text.length; } var w = document.createTreeWalker(el, NodeFilter.SHOW_TEXT); var n, seen = 0, r = document.createRange(), a = null, b = null; while ((n = w.nextNode())) { var len = n.nodeValue.length; if (!a && from <= seen + len) { a = [n, from - seen]; } if (!b && to <= seen + len) { b = [n, to - seen]; } seen += len; } r.setStart(a[0], a[1]); r.setEnd(b[0], b[1]); var s = getSelection(); s.removeAllRanges(); s.addRange(r); return s.toString(); }; window.clip = function(type, el, data){ var dt = new DataTransfer(); if (data) Object.keys(data).forEach(function(k){ dt.setData(k, data[k]); }); var ev = new ClipboardEvent(type, { clipboardData: dt, bubbles: true, cancelable: true }); el.dispatchEvent(ev); return { dt: dt, prevented: ev.defaultPrevented, html: dt.getData('text/html'), text: dt.getData('text/plain') }; }; window.card = function(id){ return document.querySelector('#comments-list .comment-card[data-id="' + id + '"]'); }; return true; })()
