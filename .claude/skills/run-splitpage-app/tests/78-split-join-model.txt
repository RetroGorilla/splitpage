# Splitting and joining scenes through the document model (#85, design
# §3.2; splitRowFieldCore in 06-rows.js, mergeWriterRows in 19-writer.js,
# syncCommentAnchors in 04c-doc-model.js).
#
#   * a split through the middle of a commented phrase leaves the comment
#     marked in both halves, and its card points at the first;
#   * a split just before the phrase carries all of it, and the card
#     follows it to the new scene;
#   * a join brings the comment back with its words, and a phrase a split
#     cut in two is one anchor again;
#   * each is ONE undo entry, and Ctrl+Z puts the words AND the comment's
#     row back.
# The pure half is src/test/comment-anchors.test.mjs. Every `eval` is one
# physical line.

open
wait document.querySelectorAll('#rows .row').length > 0

# ---- a document with one comment, on "bravo charlie" in scene 1's audio ------
eval (function(){ window.F = window.__splitpageFormats; window.U = window.__undoTest; var doc = { schema: 2, title: 'Split join model', rows: [ { id: 'r1', visualText: 'one', audioText: 'Alpha <span class="cmt-anchor" data-cmt="c1">bravo charlie</span> delta', status: 'todo' }, { id: 'r2', visualText: '', audioText: 'Echo foxtrot', status: 'todo' } ], comments: [ { id: 'c1', rowId: 'r1', field: 'audioText', anchorId: 'c1', quotedText: 'bravo charlie', author: 'Ann', body: 'about bravo', createdAt: '2026-09-01T10:00:00Z', kind: 'anchor', replies: [] } ] }; F.openFile(new File([JSON.stringify(doc)], 'split-join-model.splitpage')); return true; })()
wait F.state().title === 'Split join model' && document.querySelector('#rows .row[data-id="r2"]')
eval (function(){ window.ORIG = 'Alpha <span class="cmt-anchor" data-cmt="c1">bravo charlie</span> delta'; window.cell = function(id){ return document.querySelector('#rows .row[data-id="' + id + '"] .audio-cell .cell-editable'); }; window.para = function(id){ return document.querySelector('#writer-doc .wr-para[data-id="' + id + '"]'); }; window.aud = function(i){ return F.state().rows[i].audioText; }; window.c1 = function(){ return F.state().comments.filter(function(c){ return c.id === 'c1'; })[0]; }; window.ids = function(){ return F.state().rows.map(function(r){ return r.id; }).join(','); }; window.caretAt = function(el, offset){ el.focus(); var w = document.createTreeWalker(el, NodeFilter.SHOW_TEXT); var n, seen = 0; while ((n = w.nextNode())) { if (offset <= seen + n.nodeValue.length) break; seen += n.nodeValue.length; } var r = document.createRange(); r.setStart(n, offset - seen); r.collapse(true); var s = getSelection(); s.removeAllRanges(); s.addRange(r); return el.textContent.slice(0, offset); }; window.card = function(){ return document.querySelector('#comments-list .comment-card[data-id="c1"]'); }; window.mark = function(){ U.flush(); window.__u0 = U.sizes().undo; return window.__u0; }; return true; })()
