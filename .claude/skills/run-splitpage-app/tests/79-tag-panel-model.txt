# The tag composer and the production panel's tag operations through the
# document model (#85; 07-production-tags.js tagFieldOccurrences /
# renameTag / removeTag): labelling every mention of a phrase, renaming an
# item and removing its label are each ONE undo entry, however many cells
# they touch, and one Ctrl+Z puts back both the script and the registry.
# Also: "every mention" now finds a phrase across formatting ("red
# <strong>mug</strong>"), the words keep their casing and formatting, and a
# single mention tags exactly the occurrence that was selected.
#
# Every `eval` is one physical line (the driver splits on newlines).

open
wait document.querySelectorAll('#rows .row').length > 1
eval (function(){ window.U = window.__undoTest; window.F = window.__splitpageFormats; window.st = function (i, col) { return F.state().rows[i][col + 'Text']; }; window.snap = function () { var s = F.state(); return JSON.stringify({ rows: s.rows.map(function (r) { return [r.visualText, r.audioText]; }), reg: s.productionTags.map(function (t) { return [t.id, t.cat, t.name, t.owner, t.notes, t.status]; }) }); }; window.spans = function (name) { return F.state().rows.reduce(function (n, r) { var d = document.createElement('div'); d.innerHTML = r.visualText + r.audioText; return n + d.querySelectorAll('.tag-anchor[data-tag-name="' + name + '"]').length; }, 0); }; window.reg = function (name) { return F.state().productionTags.some(function (t) { return t.name === name; }); }; return true; })()
