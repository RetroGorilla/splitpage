# A pasted b-roll key is a line of its own (#85, src/js/05c-clipboard.js
# freshBrollIds, 08c-broll.js recordPastedBroll and the anchoring pass):
#
#   * a copy of a "B-roll:" line pasted into another scene, into the same
#     cell, or pasted a second time, gets a new id every time — never the
#     id the first line's shot is found by — and a new record: not shot, no
#     notes, seeded from the scene it landed in; the first shot keeps its
#     status and notes;
#   * the paste and its record are ONE undo entry;
#   * a cut then its paste in this tab MOVES the line: the key keeps its id
#     and the shot its status; a second paste of that cut, another tab's
#     cut, or a drop, is a copy;
#   * a key from another document whose id this one does not know keeps it;
#   * a document already holding two keys with one id (pasted by an older
#     version) is healed by the anchoring pass: the first line keeps the
#     id, the second gets its own, and the list holds one record per id.
# The pure half is src/test/clipboard.test.mjs and broll-model.test.mjs.
# Every `eval` is one physical line.

eval (function(){ window.cell = function(id, field){ return document.querySelector('#rows .row[data-id="' + id + '"] .' + (field === 'visualText' ? 'visual' : 'audio') + '-cell .cell-editable'); }; window.st = function(id, field){ return F.state().rows.filter(function(r){ return r.id === id; })[0][field]; }; window.selectText = function(el, needle, collapseEnd){ el.focus(); var text = el.textContent; var at = text.indexOf(needle); if (at === -1) throw new Error('no ' + needle); var from = at, to = at + needle.length; if (collapseEnd === 'end') { from = to = text.length; } var w = document.createTreeWalker(el, NodeFilter.SHOW_TEXT); var n, seen = 0, r = document.createRange(), a = null, b = null; while ((n = w.nextNode())) { var len = n.nodeValue.length; if (!a && from <= seen + len) { a = [n, from - seen]; } if (!b && to <= seen + len) { b = [n, to - seen]; } seen += len; } r.setStart(a[0], a[1]); r.setEnd(b[0], b[1]); var s = getSelection(); s.removeAllRanges(); s.addRange(r); return s.toString(); }; window.clip = function(type, el, data){ var dt = new DataTransfer(); if (data) Object.keys(data).forEach(function(k){ dt.setData(k, data[k]); }); var ev = new ClipboardEvent(type, { clipboardData: dt, bubbles: true, cancelable: true }); el.dispatchEvent(ev); return { dt: dt, prevented: ev.defaultPrevented, html: dt.getData('text/html'), text: dt.getData('text/plain') }; }; window.keysIn = function(html){ var out = [], re = /data-broll-id="([^"]*)"/g, m; while ((m = re.exec(html || ''))) out.push(m[1]); return out; }; window.allKeys = function(){ var out = []; F.state().rows.forEach(function(r){ out = out.concat(keysIn(r.visualText), keysIn(r.audioText)); }); return out; }; window.unique = function(list){ return list.every(function(x, i){ return list.indexOf(x) === i; }); }; window.rec = function(id){ return F.state().brollShots.filter(function(r){ return r.id === id; })[0]; }; window.recIds = function(){ return F.state().brollShots.map(function(r){ return r.id; }); }; return true; })()
